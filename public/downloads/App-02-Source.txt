const SHEET_NAME = 'Quản lý nghỉ phép';

function doGet() {
  return HtmlService.createHtmlOutputFromFile('Index')
      .setTitle('Demo Quản Lý Nghỉ Phép')
      .addMetaTag('viewport', 'width=device-width, initial-scale=1');
}

// Hàm khởi tạo Sheet và lấy Sheet active
function setupSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow(['ID', 'Họ tên', 'Từ ngày', 'Đến ngày', 'Lý do', 'Trạng thái', 'Thời gian gửi']);
    sheet.getRange("A1:G1").setFontWeight("bold");
  }
  return sheet;
}

// Lưu đơn mới
function submitLeave(data) {
  const sheet = setupSheet();
  const id = 'NP-' + new Date().getTime();
  const timestamp = new Date().toLocaleString('vi-VN');
  sheet.appendRow([id, data.name, data.startDate, data.endDate, data.reason, 'Chờ duyệt', timestamp]);
  return true;
}

// Lấy danh sách đơn
function getLeaves() {
  const sheet = setupSheet();
  const data = sheet.getDataRange().getDisplayValues();
  if (data.length <= 1) return [];
  
  let result = [];
  const headers = data[0];
  // Duyệt từ dưới lên để đơn mới nhất hiển thị trước
  for (let i = data.length - 1; i >= 1; i--) {
    let obj = { row: i + 1 }; // Lưu lại số thứ tự dòng để dễ update
    headers.forEach((h, index) => obj[h] = data[i][index]);
    result.push(obj);
  }
  return result;
}

// Cập nhật trạng thái
function updateStatus(row, newStatus) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  sheet.getRange(row, 6).setValue(newStatus); // Cột F (cột số 6) là Trạng thái
  return true;
}

    // Xử lý gửi form
    function handleFormSubmit(e) {
      e.preventDefault();
      const btn = document.getElementById('btnSubmit');
      btn.disabled = true; btn.innerText = 'Đang gửi...';
      
      const form = e.target;
      const data = {
        name: form.name.value,
        startDate: form.startDate.value,
        endDate: form.endDate.value,
        reason: form.reason.value
      };
      
      google.script.run.withSuccessHandler(() => {
        alert('Đã gửi đơn nghỉ phép thành công!');
        form.reset();
        btn.disabled = false; btn.innerText = 'Gửi Đơn Nghỉ Phép';
      }).submitLeave(data);
    }

    // Tải dữ liệu từ Google Sheets
    function loadData() {
      document.getElementById('tableBody').innerHTML = '<tr><td colspan="5" class="text-center">Đang tải...</td></tr>';
      google.script.run.withSuccessHandler(renderData).getLeaves();
    }

    // Hiển thị dữ liệu ra Bảng và Dashboard
    function renderData(data) {
      let html = '';
      let total = data.length, pending = 0, approved = 0;

      data.forEach(item => {
        // Đếm số lượng cho dashboard
        if(item['Trạng thái'] === 'Chờ duyệt') pending++;
        if(item['Trạng thái'] === 'Đã duyệt') approved++;

        // Render Table
        let badgeColor = item['Trạng thái'] === 'Chờ duyệt' ? 'bg-warning text-dark' : (item['Trạng thái'] === 'Đã duyệt' ? 'bg-success' : 'bg-danger');
        
        let buttons = item['Trạng thái'] === 'Chờ duyệt' ? 
          `<button class="btn btn-sm btn-success me-1" onclick="updateStatus(${item.row}, 'Đã duyệt', this)">Duyệt</button>
           <button class="btn btn-sm btn-outline-danger" onclick="updateStatus(${item.row}, 'Từ chối', this)">Từ chối</button>` : `<span class="text-muted small">Đã xử lý</span>`;

        html += `
          <tr>
            <td class="fw-bold">${item['Họ tên']}</td>
            <td>${item['Từ ngày']} <br>đến ${item['Đến ngày']}</td>
            <td>${item['Lý do']}</td>
            <td><span class="badge ${badgeColor}">${item['Trạng thái']}</span></td>
            <td>${buttons}</td>
          </tr>
        `;
      });
      document.getElementById('tableBody').innerHTML = html || '<tr><td colspan="5" class="text-center">Chưa có đơn nào</td></tr>';

      // Render Dashboard
      document.getElementById('dashboardStats').innerHTML = `
        <div class="col-md-4 mb-3">
          <div class="card bg-primary text-white p-4 h-100 shadow-sm">
            <h1 class="display-4 fw-bold">${total}</h1><h5>Tổng Đơn Nhận</h5>
          </div>
        </div>
        <div class="col-md-4 mb-3">
          <div class="card bg-warning text-dark p-4 h-100 shadow-sm">
            <h1 class="display-4 fw-bold">${pending}</h1><h5>Đang Chờ Duyệt</h5>
          </div>
        </div>
        <div class="col-md-4 mb-3">
          <div class="card bg-success text-white p-4 h-100 shadow-sm">
            <h1 class="display-4 fw-bold">${approved}</h1><h5>Đã Được Duyệt</h5>
          </div>
        </div>
      `;
    }

    // Hàm gọi cập nhật trạng thái khi nhấn nút Duyệt/Từ chối
    function updateStatus(row, status, btnElement) {
      btnElement.disabled = true;
      btnElement.innerText = "...";
      google.script.run.withSuccessHandler(() => loadData()).updateStatus(row, status);
    }
