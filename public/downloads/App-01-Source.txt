const SHEET_NAME = 'Incidents';

function doGet() {
  return HtmlService.createHtmlOutputFromFile('Index')
      .setTitle('Báo cáo Sự cố Nội bộ')
      .addMetaTag('viewport', 'width=device-width, initial-scale=1');
}

function setupSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow(['ID', 'Người báo cáo', 'Phòng ban', 'Thời gian xảy ra', 'Địa điểm', 'Mô tả', 'Mức độ', 'Trạng thái', 'Lịch sử', 'Cập nhật lần cuối']);
    sheet.getRange("A1:J1").setFontWeight("bold");
  }
}

function submitIncident(data) {
  setupSheet();
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  
  const id = 'INC-' + new Date().getTime();
  const status = 'Mới';
  const now = new Date().toLocaleString('vi-VN');
  const history = `[${now}] Tạo mới báo cáo\n`;
  
  sheet.appendRow([
    id, data.reporter, data.department, data.time, data.location, data.description, data.severity, status, history, now
  ]);
  return true;
}

function getIncidents() {
  setupSheet();
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  
  const headers = data[0];
  let result = [];
  for (let i = 1; i < data.length; i++) {
    let obj = {};
    headers.forEach((h, index) => obj[h] = data[i][index]);
    result.push(obj);
  }
  
  // Ưu tiên hiển thị Mức độ Nghiêm trọng lên đầu, sau đó sắp xếp theo thời gian mới nhất
  result.sort((a, b) => {
    if (a['Mức độ'] === 'Nghiêm trọng' && b['Mức độ'] !== 'Nghiêm trọng') return -1;
    if (a['Mức độ'] !== 'Nghiêm trọng' && b['Mức độ'] === 'Nghiêm trọng') return 1;
    return new Date(b['Cập nhật lần cuối']) - new Date(a['Cập nhật lần cuối']);
  });
  return result;
}

function updateStatus(id, newStatus, role) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  const data = sheet.getDataRange().getValues();
  
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === id) {
      const now = new Date().toLocaleString('vi-VN');
      let currentHistory = data[i][8] || '';
      currentHistory += `[${now}] ${role} đổi thành: ${newStatus}\n`;
      
      sheet.getRange(i + 1, 8).setValue(newStatus);
      sheet.getRange(i + 1, 9).setValue(currentHistory);
      sheet.getRange(i + 1, 10).setValue(now);
      return true;
    }
  }
  return false;
}

  <script>
    function handleFormSubmit(e) {
      e.preventDefault();
      const form = document.getElementById('incidentForm');
      const btn = document.getElementById('btnSubmit');
      btn.disabled = true;
      btn.innerText = 'Đang xử lý...';

      const data = {
        reporter: form.reporter.value,
        department: form.department.value,
        time: form.time.value,
        location: form.location.value,
        description: form.description.value,
        severity: form.severity.value
      };

      google.script.run.withSuccessHandler(() => {
        alert('Gửi báo cáo sự cố thành công!');
        form.reset();
        btn.disabled = false;
        btn.innerText = 'Gửi Báo Cáo Sự Cố';
      }).submitIncident(data);
    }

    function loadIncidents() {
      document.getElementById('loadingWorkflow').style.display = 'block';
      document.getElementById('incidentsContainer').innerHTML = '';
      google.script.run.withSuccessHandler(renderIncidents).getIncidents();
    }

    function renderIncidents(data) {
      document.getElementById('loadingWorkflow').style.display = 'none';
      const role = document.getElementById('currentRole').value;
      let html = '';
      
      data.forEach(inc => {
        const isCritical = inc['Mức độ'] === 'Nghiêm trọng' ? 'critical-incident' : '';
        const badgeColor = getStatusBadge(inc['Trạng thái']);
        
        html += `
          <div class="card mb-3 ${isCritical}">
            <div class="card-body">
              <h5 class="card-title d-flex justify-content-between">
                <span>${inc['ID']} - Mức độ: <strong>${inc['Mức độ']}</strong></span>
                <span class="badge ${badgeColor}">${inc['Trạng thái']}</span>
              </h5>
              <p class="mb-1"><strong>Phòng ban:</strong> ${inc['Phòng ban']} | <strong>Người báo cáo:</strong> ${inc['Người báo cáo']}</p>
              <p class="mb-2"><strong>Mô tả:</strong> ${inc['Mô tả']}</p>
              <div class="history-box mb-3">${inc['Lịch sử']}</div>
              <div class="actions">
                ${generateWorkflowButtons(inc['ID'], inc['Trạng thái'], role)}
              </div>
            </div>
          </div>
        `;
      });
      document.getElementById('incidentsContainer').innerHTML = html || '<p>Hiện chưa có sự cố nào.</p>';
    }

    function getStatusBadge(status) {
      switch(status) {
        case 'Mới': return 'bg-primary';
        case 'Đang xử lý': return 'bg-warning text-dark';
        case 'Chờ xác nhận': return 'bg-info text-dark';
        case 'Đã đóng': return 'bg-success';
        default: return 'bg-secondary';
      }
    }

    function generateWorkflowButtons(id, status, role) {
      if (status === 'Mới' && role === 'Quản lý') {
        return `<button class="btn btn-warning btn-sm" onclick="changeStatus('${id}', 'Đang xử lý')">Duyệt & Giao xử lý</button>`;
      }
      if (status === 'Đang xử lý') { 
        // Bất kỳ ai thực thi đều có thể báo cáo là đã xử lý xong
        return `<button class="btn btn-info btn-sm" onclick="changeStatus('${id}', 'Chờ xác nhận')">Đã xử lý (Chuyển sang Chờ xác nhận)</button>`;
      }
      if (status === 'Chờ xác nhận' && role === 'Bộ phận an toàn') {
        return `<button class="btn btn-success btn-sm" onclick="changeStatus('${id}', 'Đã đóng')">Bộ phận an toàn: Xác nhận hoàn tất & Đóng</button>`;
      }
      return `<span class="text-muted small fst-italic">Tài khoản/vai trò hiện tại không có quyền thao tác trạng thái này.</span>`;
    }

    function changeStatus(id, newStatus) {
      const role = document.getElementById('currentRole').value;
      document.getElementById('loadingWorkflow').style.display = 'block';
      document.getElementById('incidentsContainer').innerHTML = '';
      google.script.run.withSuccessHandler(() => loadIncidents()).updateStatus(id, newStatus, role);
    }

    function loadDashboard() {
      document.getElementById('loadingDashboard').style.display = 'block';
      document.getElementById('dashboardStats').innerHTML = '';
      google.script.run.withSuccessHandler(renderDashboard).getIncidents();
    }

    function renderDashboard(data) {
      document.getElementById('loadingDashboard').style.display = 'none';
      let bySeverity = {}, byDept = {}, byStatus = {};
      
      data.forEach(inc => {
        bySeverity[inc['Mức độ']] = (bySeverity[inc['Mức độ']] || 0) + 1;
        byDept[inc['Phòng ban']] = (byDept[inc['Phòng ban']] || 0) + 1;
        byStatus[inc['Trạng thái']] = (byStatus[inc['Trạng thái']] || 0) + 1;
      });

      const makeTable = (title, obj) => {
        let rows = Object.keys(obj).map(k => `<tr><td>${k}</td><td class="text-end fw-bold">${obj[k]}</td></tr>`).join('');
        return `
          <div class="col-md-4 mb-3">
            <div class="card h-100 shadow-sm border-0">
              <div class="card-header bg-dark text-white fw-bold">${title}</div>
              <div class="card-body p-0">
                <table class="table table-striped mb-0"><tbody>${rows || '<tr><td colspan="2">Chưa có dữ liệu</td></tr>'}</tbody></table>
              </div>
            </div>
          </div>
        `;
      };

      document.getElementById('dashboardStats').innerHTML = `
        <div class="col-12 mb-4">
          <div class="alert alert-success d-flex justify-content-between align-items-center">
             <h4 class="mb-0">Tổng số sự cố được ghi nhận:</h4>
             <h2 class="mb-0">${data.length}</h2>
          </div>
        </div>
        ${makeTable('Thống kê theo Mức độ', bySeverity)}
        ${makeTable('Thống kê theo Phòng ban', byDept)}
        ${makeTable('Thống kê theo Trạng thái', byStatus)}
      `;
    }
  </script>
