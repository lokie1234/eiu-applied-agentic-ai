import {useEffect,useRef,useState,type CSSProperties} from 'react';
import frame from '../../../components/section1/SectionOneHtmlSlide.module.css';
import styles from './ideas.module.css';
import ideas from './ideas-data.json';

export default function ManagerIdeas(){
  const [selected,setSelected]=useState<number|null>(null),dialog=useRef<HTMLElement>(null),trigger=useRef<HTMLElement|null>(null);
  const active=selected===null?null:ideas[selected];
  useEffect(()=>{
    if(selected===null)return;
    dialog.current?.querySelector<HTMLButtonElement>('button')?.focus();
    const key=(event:KeyboardEvent)=>{
      if(event.key==='Escape'){event.preventDefault();event.stopPropagation();setSelected(null);}
      if(event.key==='Tab'){
        const buttons=[...dialog.current?.querySelectorAll<HTMLButtonElement>('button')||[]],first=buttons[0],last=buttons[buttons.length-1];
        if(event.shiftKey&&document.activeElement===first){event.preventDefault();last?.focus();}
        else if(!event.shiftKey&&document.activeElement===last){event.preventDefault();first?.focus();}
      }
    };
    window.addEventListener('keydown',key,true);
    return()=>{window.removeEventListener('keydown',key,true);if(trigger.current?.isConnected)trigger.current.focus();};
  },[selected]);
  return <section className={`${frame.root} ${styles.root}`} data-manager-ideas>
    <div className={frame.ambient} aria-hidden="true"><div className={frame.grid}/><div className={frame.glow}/><svg viewBox="0 0 960 540"><path d="M-45 430 C155 318 300 455 474 298 C640 150 770 235 1010 85"/></svg></div>
    <header className={frame.heading}><span>BẮT ĐẦU TỪ NHU CẦU</span><h1>Một vài ý tưởng cho cấp quản lý</h1></header>
    <div className={styles.grid} data-idea-grid>{ideas.map((idea,index)=><button key={idea.id} type="button" className={styles.idea} data-manager-idea={idea.id} aria-label={idea.title} aria-haspopup="dialog" onClick={event=>{trigger.current=event.currentTarget;setSelected(index);}}>
      <span className={styles.tileIcon} data-idea-tile-icon><Glyph kind={idea.icon}/></span><strong>{idea.title}</strong>
    </button>)}</div>
    {active&&<div className={styles.backdrop} onClick={event=>{if(event.target===event.currentTarget)setSelected(null);}}>
      <section className={styles.dialog} role="dialog" aria-modal="true" aria-label={`Quy trình · ${active.title}`} ref={dialog} data-idea-dialog={active.id}>
        <header className={styles.dialogHeader} data-idea-title><span className={styles.dialogHeadingIcon}><Glyph kind={active.icon}/></span><div className={styles.dialogTitle}><h2>{active.title}</h2><span className={styles.titleAccent} aria-hidden="true"/></div><button type="button" className={styles.close} aria-label="Đóng quy trình" onClick={()=>setSelected(null)}>×</button></header>
        <div className={styles.flow} aria-label="Quy trình minh họa">{active.steps.map((step,index)=><div key={active.id+'-'+index} className={styles.step} style={{'--step-delay':`${index*1.6}s`} as CSSProperties} data-idea-step>
          <span className={styles.stepNumber}>0{index+1}</span><div className={styles.stepIcon}><Glyph kind={step.icon}/><span className={styles.complete}>✓</span></div><h3>{step.label}</h3>
          {index<3&&<span className={styles.connector} aria-hidden="true"><i/><b>›</b></span>}
        </div>)}</div>
        <footer className={styles.dialogFooter}><span className={styles.result}><i/> {active.result}</span><div><button type="button" aria-label="Ý tưởng trước" onClick={()=>setSelected((selected!+ideas.length-1)%ideas.length)}>←</button><span>{String(selected!+1).padStart(2,'0')} / 08</span><button type="button" aria-label="Ý tưởng tiếp theo" onClick={()=>setSelected((selected!+1)%ideas.length)}>→</button></div></footer>
      </section>
    </div>}
  </section>;
}
