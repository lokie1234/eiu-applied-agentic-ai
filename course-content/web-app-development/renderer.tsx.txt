function vscodeMap(s) {
    const host = h('div', { className: 'vscodeMap workspaceMapInteractive' });
    let overlay = null, opener = null;
    function close() { if (!overlay)
        return; overlay.remove(); overlay = null; opener?.focus(); }
    function open(index, button) {
        const detail = s.regionDetails?.[index];
        if (!detail)
            return;
        close();
        opener = button;
        const r = vscodeRegions[index];
        const closeButton = h("button", { type: "button", className: "workspacePopupClose", "aria-label": 'Đóng ' + r.name, onClick: close }, "\u0110\u00F3ng \u00D7");
        const panel = h("section", { className: "workspacePopup", role: "dialog", "aria-modal": "true", "aria-labelledby": "workspace-popup-title" },
            h("header", { className: "workspacePopupHeader" },
                h("span", { className: "workspacePopupNumber", "aria-hidden": "true" }, String(index + 1).padStart(2, '0')),
                h("div", { className: "workspacePopupHeading" },
                    h("small", null,
                        "V\u00D9NG ",
                        String(index + 1).padStart(2, '0'),
                        " \u00B7 VS CODE"),
                    h("h2", { id: "workspace-popup-title" }, detail.title),
                    h("p", null, detail.message)),
                closeButton),
            regionDetail(detail));
        overlay = h('div', { className: 'workspacePopupOverlay' });
        overlay.addEventListener('click', (event) => { if (event.target === overlay)
            close(); });
        overlay.addEventListener('keydown', (event) => {
            if (event.key === 'Escape') {
                event.preventDefault();
                event.stopPropagation();
                close();
            }
            if (event.key === 'Tab') {
                const items = Array.from(panel.querySelectorAll('button,[tabindex="0"]'));
                const first = items[0], last = items[items.length - 1];
                if (event.shiftKey && document.activeElement === first) {
                    event.preventDefault();
                    last.focus();
                }
                else if (!event.shiftKey && document.activeElement === last) {
                    event.preventDefault();
                    first.focus();
                }
            }
        });
        overlay.append(panel);
        host.append(overlay);
        closeButton.focus();
    }
    const shot = h("div", { className: "vscodeShot" },
        h("img", { src: assets['VS Code workspace'], alt: "\u1EA2nh VS Code c\u1EE7a d\u1EF1 \u00E1n, v\u1EDBi s\u00E1u v\u00F9ng \u0111\u01B0\u1EE3c \u0111\u00E1nh s\u1ED1 v\u00E0 khoanh \u0111\u1ECF" }),
        h("svg", { className: "vscodeMarks", viewBox: "0 0 1917 1152", "aria-hidden": "true" }, vscodeRegions.map((r, i) => h("g", { className: "vscodeRegionReveal", style: { animationName: 'vscodeRegionReveal' + (i + 1) } },
            h("rect", { x: r.x, y: r.y, width: r.w, height: r.h, rx: "14" }),
            h("circle", { cx: r.cx, cy: r.cy, r: "29" }),
            h("text", { x: r.cx, y: r.cy }, i + 1)))));
    vscodeRegions.forEach((r, i) => { const target = h("button", { type: "button", className: "vscodeHotspot", "aria-haspopup": "dialog", "aria-label": 'Xem vùng ' + (i + 1) + ': ' + r.name, style: { left: r.x / 1917 * 100 + '%', top: r.y / 1152 * 100 + '%', width: r.w / 1917 * 100 + '%', height: r.h / 1152 * 100 + '%' }, onClick: () => open(i, target) }); shot.append(target); });
    host.append(h("style", null, popupTypographyCss + workspacePopupCss), shot);
    const names = h('div', { className: 'vscodeNames' });

        h("span", { className: "vscodeBadge" }, i + 1),
        h("div", null,
            h("h3", null, r.name))); names.append(button); });
    host.append(names);
    return host;
}

let editableGitId = 0;
function editableGitCommands(mode) {
    const id = 'editable-git-' + (++editableGitId), fields = [];
    const host = h('div', { className: 'editableGit', 'data-mode': mode });
    const grid = h('div', { className: 'editableGitFields' });
    const preview = h("pre", { className: "editableGitPreview", "aria-label": "L\u1EC7nh theo d\u1EEF li\u1EC7u \u0111\u00E3 nh\u1EADp" });
    const status = h("span", { className: "editableGitStatus", id: id + '-status', "aria-live": "polite" });
    const copySlot = h('div', { className: 'editableGitCopy' });
    const specs = mode === 'identity' ? [['Tên tác giả Commit', 'Nguyễn Văn An', 'text'], ['Email GitHub', 'ban@example.com', 'email']] : [['Địa chỉ HTTPS của repository', 'https://github.com/USERNAME/REPO.git', 'url']];
    const quote = (value) => "'" + value.replace(/'/g, "''") + "'";
    function refresh() {
        const values = fields.map(field => field.value.trim());
        const invalid = values.map((value, i) => !value || /[\r\n\x00-\x1f\x7f]/.test(value) || (mode === 'identity' && i === 1 && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value)));
        let repo = '';
        if (mode === 'repo' && values[0]) {
            try {
                const url = new URL(values[0]);
                if (url.protocol !== 'https:' || url.hostname !== 'github.com' || url.port || url.username || url.password || url.search || url.hash || !/^\/[A-Za-z0-9_-]+\/[A-Za-z0-9_.-]+\/?$/.test(url.pathname))
                    throw Error('Invalid repo');
                repo = url.href.replace(/\/$/, '');
            }
            catch {
                invalid[0] = true;
            }
        }
        fields.forEach((field, i) => field.setAttribute('aria-invalid', String(Boolean(values[i]) && invalid[i])));
        const valid = !invalid.some(Boolean);
        const command = mode === 'identity' ? ['git config --global user.name ' + quote(values[0] || '<Tên của bạn>'), 'git config --global user.email ' + quote(values[1] || '<Email của bạn>'), 'git config --global init.defaultBranch main'].join('\n') : ['git remote add origin ' + (repo || 'https://github.com/USERNAME/REPO.git'), 'git remote -v'].join(' && ');
        preview.textContent = command;
        status.textContent = valid || (mode === 'repo' && !values.some(Boolean)) ? '' : mode === 'identity' ? 'Nhập đủ tên và email hợp lệ để sao chép.' : 'Dán URL HTTPS dạng github.com/owner/repo để sao chép.';
        status.dataset.error = String(values.some(Boolean) && invalid.some(Boolean));
        // Each rendered button captures this exact validated snapshot; edits create a fresh one.
        const copy = localCopyButton(command);
        copy.disabled = !valid;
        copy.setAttribute('aria-describedby', id + '-status');
        copySlot.replaceChildren(copy);
    }
    specs.forEach(([label, placeholder, type], i) => { const input = h("input", { id: id + '-' + i, type: type, placeholder: placeholder, autoComplete: "off", spellCheck: false, "aria-describedby": id + '-status', onInput: refresh }); fields.push(input); grid.append(h("label", { htmlFor: input.id },
        label,
        input)); });
    host.append(h("style", null, editableGitCss), grid, preview, h("div", { className: "editableGitActions" },
        status,
        copySlot));
    if (mode === 'repo')
        host.append(h("div", { className: "repoSuccessExample" },
            h("strong", null, "V\u00ED d\u1EE5 k\u1EBFt n\u1ED1i th\u00E0nh c\u00F4ng"),
            h("pre", null, 'origin  https://github.com/USERNAME/REPO.git (fetch)\norigin  https://github.com/USERNAME/REPO.git (push)')));
    refresh();
    return host;
}

function editableCloneCommands() {
    const id = 'editable-clone-' + (++editableGitId);
    const host = h("div", { className: "editableGit cloneEditor", "data-mode": "clone" },
        h("style", null, editableGitCss + cloneEditorCss));
    const preview = h("pre", { className: "editableGitPreview", "aria-label": "L\u1EC7nh Clone theo d\u1EEF li\u1EC7u \u0111\u00E3 nh\u1EADp" });
    const status = h("span", { className: "editableGitStatus", id: id + '-status', "aria-live": "polite" });
    const copySlot = h("div", { className: "editableGitCopy" });
    const repo = h("input", { id: id + '-url', type: "url", placeholder: "https://github.com/owner/repo.git", autoComplete: "off", autocorrect: "off", autocapitalize: "off", "aria-describedby": id + '-status', onInput: refresh });
    const folder = h("input", { id: id + '-folder', type: "text", value: "eiu-advertising-starter", autoComplete: "off", autocorrect: "off", autocapitalize: "off", "aria-describedby": id + '-status', onInput: refresh });
    for (const field of [repo, folder]) {
        field.spellcheck = false;
        field.setAttribute('spellcheck', 'false');
    }
    const quote = (text) => "'" + text.replace(/'/g, "''") + "'";
    function refresh() {
        let url = '';
        const raw = repo.value.trim(), dir = folder.value;
        try {
            const parsed = new URL(raw);
            if (parsed.protocol !== 'https:' || parsed.hostname !== 'github.com' || parsed.port || parsed.username || parsed.password || parsed.search || parsed.hash || /[\x00-\x20\x7f]/.test(raw) || !/^\/[A-Za-z0-9_-]+\/[A-Za-z0-9_.-]+\/?$/.test(parsed.pathname))
                throw Error('Invalid URL');
            url = parsed.href.replace(/\/$/, '');
        }
        catch { }
        const folderValid = Boolean(dir.trim()) && dir === dir.trim() && !/[<>:"/\\|?*\[\]\x00-\x1f\x7f]/.test(dir) && !/[. ]$/.test(dir) && !/^\.{1,2}$/.test(dir) && !/^(con|prn|aux|nul|com[1-9]|lpt[1-9])(?:\.|$)/i.test(dir);
        const valid = Boolean(url) && folderValid;
        repo.setAttribute('aria-invalid', String(Boolean(raw) && !url));
        folder.setAttribute('aria-invalid', String(!folderValid));
        const command = ['git clone -- ' + (url ? quote(url) : 'URL_REPO') + ' ' + quote(dir || '<thu-muc>'), 'cd ' + quote(dir || '<thu-muc>'), 'git remote -v', 'npm install', 'npm run dev'].join('\n');
        preview.textContent = command;
        status.textContent = !folderValid ? 'Nhập tên thư mục hợp lệ.' : raw && !url ? 'URL GitHub HTTPS không hợp lệ.' : '';
        status.dataset.error = String(Boolean(raw) && !url || !folderValid);
        const button = localCopyButton(command);
        button.disabled = !valid;
        button.setAttribute('aria-describedby', id + '-status');
        copySlot.replaceChildren(button);
    }
    host.append(h("div", { className: "editableGitFields" },
        h("label", { htmlFor: repo.id },
            "URL GitHub c\u1EE7a b\u1EA1n",
            repo),
        h("label", { htmlFor: folder.id },
            "T\u00EAn th\u01B0 m\u1EE5c Clone / cd",
            folder)), preview, h("div", { className: "editableGitActions" },
        status,
        copySlot));
    refresh();
    return host;
}

function openStarterPrompt(trigger, prompt) {
    const host = trigger.closest('.eiu');
    if (!host)
        return;
    let overlay;
    const close = () => { overlay.remove(); if (trigger.isConnected)
        trigger.focus(); };
    const closeButton = h("button", { type: "button", className: "starterDownloadClose", "aria-label": "\u0110\u00F3ng prompt", onClick: close }, "\u00D7");
    const copy = localCopyButton(prompt);
    const dialog = h("section", { className: "starterDownloadDialog starterPromptDialog", role: "dialog", "aria-modal": "true", "aria-labelledby": "starter-prompt-title" },
        h("header", null,
            h("h2", { id: "starter-prompt-title" }, "Prompt t\u1EA1o starter project"),
            copy,
            closeButton),
        h("pre", { tabIndex: 0 }, prompt));
    overlay = h("div", { className: "starterDownloadOverlay", onClick: (event) => { if (event.target === overlay)
            close(); }, onKeyDown: (event) => { if (event.key === 'Escape') {
            event.preventDefault();
            event.stopPropagation();
            close();
        }
        else if (event.key === 'Tab') {
            const controls = Array.from(dialog.querySelectorAll('button,[tabindex="0"]'));
            const first = controls[0], last = controls[controls.length - 1];
            if (event.shiftKey && document.activeElement === first) {
                event.preventDefault();
                last.focus();
            }
            else if (!event.shiftKey && document.activeElement === last) {
                event.preventDefault();
                first.focus();
            }
        } } }, dialog);
    host.append(overlay);
    closeButton.focus();
}

function starterFallingIcons(names) {
    const reduced = window.matchMedia('(prefers-reduced-motion: reduce)');
    const size = names ? 44 : 32, columns = names ? 4 : 8, spacing = names ? 60 : 49;
    const tools = names ? names.map(name => [name, name]) : [['VS Code', 'VS Code'], ['Claude', 'AI Claude Code'], ['Codex', 'AI Codex'], ['Antigravity', 'AI Antigravity'], ...aiMarqueeNames.map(name => [name, 'AI ' + name])];
    const bodies = tools.map(([name, key], i) => ({ node: h("span", { className: names ? 'vibePromptTool ' + (name === 'GitHub' ? 'github' : '') : 'starterToolIcon' }, names ? gitGuideMark(name) : h("img", { src: assets[key], alt: name })), x: 6 + (i % columns) * spacing + Math.random() * 5, y: 4 + Math.floor(i / columns) * 34, vx: (Math.random() - .5) * 90, vy: 0, angle: (Math.random() - .5) * 30, spin: (Math.random() - .5) * 140, lastHit: -1000 }));
    const arena = h("div", { className: names ? 'vibePromptTools' : 'starterToolIcons', "aria-label": "C\u00F4ng c\u1EE5 ph\u00E1t tri\u1EC3n c\u00F9ng AI" }, bodies.map(body => body.node));
    let running = false, lastTime = 0, elapsed = 0;
    const paint = () => bodies.forEach(body => { body.node.style.transform = `translate(${body.x}px,${body.y}px) rotate(${body.angle}deg)`; });
    const rest = () => { bodies.forEach((body, i) => { body.x = 6 + (i % columns) * spacing; body.y = 8 + Math.floor(i / columns) * 36; body.angle = 0; body.vx = body.vy = body.spin = 0; }); paint(); };
    function frame(now) {
        if (!arena.isConnected) {
            running = false;
            return;
        }
        if (reduced.matches) {
            rest();
            running = false;
            return;
        }
        const dt = lastTime ? Math.min((now - lastTime) / 1000, .032) : .016;
        lastTime = now;
        elapsed += dt;
        const right = Math.max(0, (arena.clientWidth || (names ? 260 : 420)) - size - 4), floor = Math.max(0, (arena.clientHeight || 88) - size - 4);
        let moving = false;
        for (const body of bodies) {
            body.vy += 900 * dt;
            body.x += body.vx * dt;
            body.y += body.vy * dt;
            body.angle += body.spin * dt;
            if (body.x < 4) {
                body.x = 4;
                body.vx = Math.abs(body.vx) * .65;
            }
            else if (body.x > right) {
                body.x = right;
                body.vx = -Math.abs(body.vx) * .65;
            }
            if (body.y < 4) {
                body.y = 4;
                body.vy = Math.abs(body.vy) * .2;
            }
            if (body.y >= floor) {
                body.y = floor;
                body.vy = Math.abs(body.vy) > 45 ? -body.vy * .48 : 0;
                body.vx *= Math.pow(.12, dt);
                body.spin *= Math.pow(.04, dt);
                if (Math.abs(body.vx) < 2)
                    body.vx = 0;
                if (Math.abs(body.spin) < 2)
                    body.spin = 0;
            }
            if (body.y < floor || Math.abs(body.vy) > 1 || Math.abs(body.vx) > 1 || Math.abs(body.spin) > 1)
                moving = true;
        }
        // Separate touching circles so the larger collection stays visible when it settles.
        for (let pass = 0; pass < 3; pass++)
            for (let i = 0; i < bodies.length; i++)
                for (let j = i + 1; j < bodies.length; j++) {
                    const a = bodies[i], b = bodies[j], dx = b.x - a.x, dy = b.y - a.y, distance = Math.hypot(dx, dy);
                    if (distance >= size + 1)
                        continue;
                    const nx = distance ? dx / distance : 1, ny = distance ? dy / distance : 0, overlap = (size + 1 - distance) / 2;
                    a.x -= nx * overlap;
                    a.y -= ny * overlap;
                    b.x += nx * overlap;
                    b.y += ny * overlap;
                    const speed = (b.vx - a.vx) * nx + (b.vy - a.vy) * ny;
                    if (speed < 0) {
                        const impulse = -speed * .6;
                        a.vx -= impulse * nx;
                        a.vy -= impulse * ny;
                        b.vx += impulse * nx;
                        b.vy += impulse * ny;
                    }
                    a.x = Math.max(4, Math.min(right, a.x));
                    b.x = Math.max(4, Math.min(right, b.x));
                    a.y = Math.max(4, Math.min(floor, a.y));
                    b.y = Math.max(4, Math.min(floor, b.y));
                }
        paint();
        if (moving && elapsed < 5)
            window.requestAnimationFrame(frame);
        else {
            bodies.forEach(body => { body.vx = body.vy = body.spin = 0; });
            running = false;
        }
    }
    const start = () => { if (running || !arena.isConnected)
        return; if (reduced.matches) {
        rest();
        return;
    } running = true; lastTime = 0; elapsed = 0; window.requestAnimationFrame(frame); };
    for (const body of bodies) {
        const launch = () => { if (reduced.matches)
            return; const now = Date.now(); if (now - body.lastHit < 200)
            return; body.lastHit = now; body.vy = -260 - Math.random() * 120; body.vx = (Math.random() < .5 ? -1 : 1) * (90 + Math.random() * 120); body.spin = (Math.random() - .5) * 450; elapsed = 0; start(); };
        body.node.addEventListener('pointerenter', launch);
        body.node.addEventListener('pointerdown', launch);
    }
    paint();
    setTimeout(start, 0);
    return arena;
}
