function renderRootCauseProcess(content, process) {
  content.dataset.mis311Process = 'true';
  content.style.overflow = 'auto';
  const host = document.defaultView;
  const requestId = host?.eiuLessonRequestId;
  let ready = !requestId;
  let epoch = '';
  let state = { selected: 0, symptom: [false,false,false], rootCause: [false,false,false], processStep: 0 };
  const style = document.createElement('style');
  style.textContent = `
    [data-mis311-process]{display:flex;flex-direction:column;gap:clamp(12px,1.2vw,18px);font-family:Arial,sans-serif;color:#17324a;scrollbar-width:thin}
    .mis311-process-title{margin:0;color:#144069;font-size:clamp(24px,2.5vw,36px);line-height:1.18;text-align:center;text-wrap:balance}
    .mis311-process-flow{flex:1;min-height:0;display:grid;grid-template-rows:repeat(7,minmax(0,1fr));gap:10px;list-style:none;padding:0;margin:0}
    .mis311-process-step{position:relative;min-width:0;min-height:0;display:grid;grid-template-columns:minmax(230px,30%) minmax(0,1fr);align-items:center;gap:18px;border:1px solid #d4e0e8;border-radius:12px;background:linear-gradient(105deg,#f3f7fa,#fff);padding:8px 16px;box-shadow:0 2px 8px #14406908}
    .mis311-process-heading{display:flex;align-items:center;gap:12px;min-width:0}
    .mis311-process-number{flex-shrink:0;display:grid;place-items:center;width:30px;height:30px;border-radius:50%;background:#144069;color:white;font-size:15px;font-weight:700}
    .mis311-process-step h2{margin:0;color:#144069;font-size:clamp(16px,1.5vw,21px);line-height:1.2;text-wrap:pretty}
    .mis311-process-step ul{padding-left:1.1em;margin:0;display:grid;gap:2px;font-size:clamp(15px,1.35vw,19px);line-height:1.35;overflow-wrap:break-word}
    .mis311-process-step li::marker{color:#b39159}
    .mis311-process-step:last-child{border-color:#abd0d8;background:linear-gradient(105deg,#eaf5f6,#fff)}
    .mis311-process-step:last-child .mis311-process-number{background:#236879}
    .mis311-process-step:last-child ul>li{text-wrap:balance}
    .mis311-process-arrow{position:absolute;display:grid;place-items:center;left:30px;top:100%;height:10px;color:#9d7944;font-size:14px;line-height:1;z-index:1}
    .mis311-process-arrow span{display:none}.mis311-process-arrow::after{content:'\\2193'}
    @media(max-width:1000px),(max-height:600px){
      [data-mis311-process][data-mode=overview]{display:block;overscroll-behavior:contain}
      .mis311-process-title{font-size:24px;margin-bottom:18px}
      .mis311-process-flow{display:flex;flex-direction:column;gap:22px;padding-bottom:2px}
      .mis311-process-step{padding:14px 16px;grid-template-columns:1fr;gap:10px;flex:none}
      .mis311-process-step h2{font-size:19px}.mis311-process-step ul{font-size:16px;line-height:1.45}
      .mis311-process-arrow{height:22px;font-size:19px}
    }
    [data-mis311-process][data-mode=intro]{justify-content:center;cursor:pointer}
    [data-mis311-process][data-mode=intro] .mis311-process-title{font-size:clamp(28px,3.3vw,48px);padding:24px}
    [data-mis311-process][data-mode=focus]{cursor:pointer}
    [data-mis311-process][data-mode=focus] .mis311-process-flow{min-height:min-content;display:flex;align-items:center;justify-content:center;padding:14px}
    [data-mis311-process][data-mode=focus] .mis311-process-step{display:flex;flex-direction:column;align-items:stretch;gap:24px;width:min(92%,980px);padding:clamp(24px,3.2vw,44px);border-radius:22px;animation:mis311-process-enter .3s ease both}
    [data-mis311-process][data-mode=focus] .mis311-process-heading{gap:18px}
    [data-mis311-process][data-mode=focus] .mis311-process-number{width:48px;height:48px;font-size:25px}
    [data-mis311-process][data-mode=focus] h2{font-size:clamp(23px,2.7vw,38px);line-height:1.25}
    [data-mis311-process][data-mode=focus] ul{font-size:clamp(20px,2.25vw,31px);line-height:1.5;gap:12px}
    [data-mis311-process][data-mode=focus] .mis311-process-arrow{display:none}
    [data-mis311-process][data-mode=overview] .mis311-process-step{animation:mis311-process-pulse 14s linear var(--mis311-process-delay) infinite}
    [data-mis311-process]:focus-visible{outline:3px solid #b39159;outline-offset:-3px}
    [data-mis311-process] [hidden]{display:none!important}
    @media(max-width:700px){
      [data-mis311-process][data-mode=focus] .mis311-process-flow{padding:12px 0}
      [data-mis311-process][data-mode=focus] .mis311-process-step{width:100%;padding:22px;gap:18px}
      [data-mis311-process][data-mode=focus] .mis311-process-number{width:36px;height:36px;font-size:20px}
      [data-mis311-process][data-mode=focus] .mis311-process-heading{gap:12px}
      [data-mis311-process][data-mode=focus] h2{font-size:23px}
      [data-mis311-process][data-mode=focus] ul{font-size:20px}
    }
    @keyframes mis311-process-enter{from{opacity:0;transform:translateY(12px)}to{opacity:1;transform:translateY(0)}}
    @keyframes mis311-process-pulse{
      0%,14.2857%,100%{box-shadow:0 2px 8px #14406908}
      3%,10.5%{box-shadow:inset 0 0 0 2px #b39159aa,0 0 14px #b3915966,0 0 4px #e9ca8b88}
    }
    @media(prefers-reduced-motion:reduce){.mis311-process-step{animation:none}}
  `;
  const title = document.createElement('h1');
  title.className = 'mis311-process-title';
  title.textContent = process.title;
  const flow = document.createElement('ol');
  flow.className = 'mis311-process-flow';
  flow.id = 'mis311-process-steps';
  flow.setAttribute('aria-label', 'Root cause analysis: seven steps in order');
  process.steps.forEach((step, index) => {
    const card = document.createElement('li');
    card.className = 'mis311-process-step';
    card.style.setProperty('--mis311-process-delay', `${index * 2}s`);
    const heading = document.createElement('div');
    heading.className = 'mis311-process-heading';
    const number = document.createElement('span');
    number.className = 'mis311-process-number';
    number.textContent = String(index + 1);
    number.setAttribute('aria-hidden', 'true');
    const label = document.createElement('h2');
    label.textContent = step.title;
    heading.append(number, label);
    const bullets = document.createElement('ul');
    for (const text of step.bullets) {
      const bullet = document.createElement('li');
      bullet.textContent = text;
      bullets.appendChild(bullet);
    }
    card.append(heading, bullets);
    if (index < process.steps.length - 1) {
      const arrow = document.createElement('span');
      arrow.className = 'mis311-process-arrow';
      arrow.setAttribute('aria-hidden', 'true');
      const glyph = document.createElement('span');
      glyph.textContent = index === 3 ? '\u2193' : index > 3 ? '\u2190' : '\u2192';
      arrow.appendChild(glyph);
      card.appendChild(arrow);
    }
    flow.appendChild(card);
  });
  content.replaceChildren(style, title, flow);
  const cards = [...flow.children];
  const step = () => state.processStep || 0;
  function update() {
    const current = step();
    content.dataset.mode = current === 0 ? 'intro' : current === 8 ? 'overview' : 'focus';
    content.dataset.processStep = String(current);
    flow.hidden = current === 0;
    cards.forEach((card, index) => { card.hidden = current !== 8 && index !== current - 1; });
    content.setAttribute('role', current === 8 ? 'group' : 'button');
    content.tabIndex = current === 8 ? -1 : 0;
    content.setAttribute('aria-controls', flow.id);
    content.setAttribute('aria-disabled', String(!ready || current === 8));
    content.setAttribute('aria-label', current === 0 ? process.title : current === 8 ? process.title : `${process.title}. Step ${current}. ${process.steps[current - 1].title}. ${process.steps[current - 1].bullets.join(' ')}`);
  }
  function advance() {
    if (!ready || step() === 8 || host?.eiuSectionEnded) return;
    const wasIntro = step() === 0;
    if(!wasIntro)title.getAnimations?.().forEach(animation=>animation.cancel());
    const before = title.getBoundingClientRect().top;
    state.processStep = step() + 1;
    update();
    content.scrollTop = 0;
    if (wasIntro && typeof title.animate === 'function' && !host?.matchMedia('(prefers-reduced-motion:reduce)').matches) {
      const distance = before - title.getBoundingClientRect().top;
      if (distance > 0) title.animate([{transform:`translateY(${distance}px)`},{transform:'translateY(0)'}], {duration:450,easing:'cubic-bezier(.22,1,.36,1)'});
    }
    if (requestId && epoch) host.parent.postMessage({type:'eiu-reveal-save',requestId,epoch,state},'*');
  }
  content.addEventListener('click', advance);
  content.addEventListener('keydown', event => { if (event.target === content && (event.key === 'Enter' || event.key === ' ')) { event.preventDefault(); advance(); } });
  if (requestId) {
    const receive = event => {
      const message = event.data;
      if (event.source !== host.parent || message?.type !== 'eiu-reveal-state' || message.requestId !== requestId || host.eiuSectionEnded) return;
      const value = message.state;
      if (!value || typeof message.epoch !== 'string' || (value.processStep !== undefined && (!Number.isInteger(value.processStep) || value.processStep < 0 || value.processStep > 8))) return;
      state = value; epoch = message.epoch; ready = true; update();
    };
    const refresh=()=>{ready=false;update();};
    host.addEventListener('eiu-reveal-refresh',refresh);
    host.addEventListener('message', receive);
    host.addEventListener('pagehide', () => {host.removeEventListener('message', receive);host.removeEventListener('eiu-reveal-refresh',refresh);}, {once:true});
    host.parent.postMessage({type:'eiu-reveal-load',requestId},'*');
  }
  update();
  content.parentElement.setAttribute('role', 'group');
}
