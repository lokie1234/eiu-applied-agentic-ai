function renderTkiDetail(content,config,kind){
  content.dataset.mis311Conflict=kind;
  const element=(tag,className,text)=>{const node=document.createElement(tag);if(className)node.className=className;if(text!==undefined)node.textContent=text;return node;};
  const button=(label,callback)=>{const node=element('button','mis311-tki-button',label);node.type='button';node.addEventListener('click',callback);return node;};
  const style=conflictSectionStyle();const title=element('h1',kind==='steps'?'mis311-tki-title':'mis311-tki-title mis311-tki-problem',kind==='steps'?config.howTitle:config.problem);const details=element('div','mis311-tki-details');
  const pages=element('nav','mis311-tki-pages');pages.setAttribute('aria-label',kind==='steps'?'TKI steps':'Example situations');const view=element('section','mis311-tki-detail-view');const buttons=[];
      const cachedPages=new Map();
      function select(index){if(view.dataset.tkiPage===String(index+1))return;view.dataset.tkiPage=String(index+1);const cached=cachedPages.get(index);if(cached)view.replaceChildren(...cached);else{view.replaceChildren();if(kind==='steps'){const step=config.steps[index];view.appendChild(element('h3','mis311-tki-step-heading',step.title));if(step.choices){const list=element('ul','mis311-tki-choices');step.choices.forEach(text=>list.appendChild(element('li','',text)));view.appendChild(list);}else view.appendChild(element('p','mis311-tki-step-text',step.text));}else{const item=config.cases[index];const card=element('div','mis311-tki-case');card.append(element('h3','',item.condition),element('strong','',`Use ${item.style}`),element('p','',item.quote));view.append(card,element('p','mis311-tki-conclusion',config.conclusion));}cachedPages.set(index,[...view.childNodes]);}buttons.forEach((button,i)=>{if(index===i)button.setAttribute('aria-current','step');else button.removeAttribute('aria-current');});}

  content.replaceChildren(style,header,details);content.parentElement.setAttribute('role','group');
}
function renderConflictHub(content,hub){
  content.dataset.mis311ConflictHub='true';
  renderRootCauseHub(content,hub,{
    model:child=>renderConflictSection(child,hub.model),
    how:child=>renderTkiDetail(child,hub.model,'steps'),
    example:child=>renderTkiDetail(child,hub.model,'example')
  });
}
