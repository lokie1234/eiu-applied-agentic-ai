  `;

  const story=document.createElement('div');story.className='mis311-defect-story';
  [['Objective:',exercise.objective],['Scenario',exercise.scenario]].forEach(([label,text])=>{const section=document.createElement('section');const heading=document.createElement('h2');heading.textContent=label;const paragraph=document.createElement('p');paragraph.textContent=text;section.append(heading,paragraph);story.appendChild(section);});

  const diagram=document.createElement('div');diagram.className='mis311-defect-diagram';
  const ns='http://www.w3.org/2000/svg';
  const svg=document.createElementNS(ns,'svg');svg.setAttribute('viewBox','0 0 1280 420');svg.setAttribute('role','group');svg.setAttribute('aria-label',exercise.diagramTitle);
  function node(tag,attrs,text){const element=document.createElementNS(ns,tag);Object.entries(attrs).forEach(([name,value])=>element.setAttribute(name,String(value)));if(text!==undefined)element.textContent=text;return element;}
  const axis=node('g',{'data-defect-axis':'true'});
  axis.appendChild(node('path',{d:'M25 210H1120m-12-8 12 8-12 8',class:'mis311-defect-spine'}));
  axis.appendChild(node('rect',{x:1130,y:179,width:140,height:62,rx:12,fill:'#144069'}));
  const effect=node('text',{x:1200,y:204,'text-anchor':'middle',fill:'#fff','font-size':19,'font-family':'Arial','font-weight':700});effect.append(node('tspan',{x:1200},'High defect'),node('tspan',{x:1200,dy:24},'rate'));axis.appendChild(effect);svg.appendChild(axis);
  const mobile=document.createElement('div');mobile.className='mis311-defect-mobile';
  const entries=[];
  exercise.groups.forEach((item,index)=>{
    const top=index<3,col=index%3,x=top?30+col*350:180+col*350,y=top?130:258;
    const center=x+125;
    const branch=node('path',{d:top?`M${center} ${y+40}L${center+60} 210`:`M${center-60} 210L${center} ${y}`,class:'mis311-defect-branch'});svg.appendChild(branch);
    const causes=node('g',{id:`mis311-defect-causes-${index}`,class:'mis311-defect-causes',hidden:'true','aria-hidden':'true'});
    item.causes.forEach((text,j)=>{
      const textY=top?28+j*32:332+j*32;
      causes.appendChild(node('path',{d:top?`M${center} ${y}H${x+12}V${textY-6}H${x+23}`:`M${center} ${y+40}H${x+12}V${textY-6}H${x+23}`,class:'mis311-defect-cause-line'}));
      causes.appendChild(node('text',{x:x+28,y:textY,class:'mis311-defect-cause'},text));
    });
    const trigger=node('g',{class:'mis311-defect-category',role:'button',tabindex:0,'aria-label':item.label,'aria-expanded':'false','aria-controls':causes.id,'data-defect-group':index});
    trigger.append(node('rect',{x,y,width:250,height:40,rx:11}),node('text',{x:center,y:y+27,'text-anchor':'middle'},item.label));
    svg.append(causes,trigger);
    const section=document.createElement('section');section.className='mis311-defect-mobile-section';

    item.causes.forEach(text=>{const item=document.createElement('li');item.textContent=text;list.appendChild(item);});section.append(button,list);mobile.appendChild(section);
    const open=event=>{event?.stopPropagation();if(!ready||stage()<5||host?.eiuSectionEnded)return;state.exerciseGroups??=[false,false,false,false,false,false];if(state.exerciseGroups[index])return;state.exerciseGroups[index]=true;update();save();};
    trigger.addEventListener('click',open);trigger.addEventListener('keydown',event=>{if(event.key==='Enter'||event.key===' '){event.preventDefault();open();}});button.addEventListener('click',open);
    entries.push({trigger,causes,button,list,branch});
  });

  const stage=()=>Math.max(state.exerciseStage||0,state.exerciseGroups?.some(Boolean)?5:0);
  function update(){
    const current=stage();content.dataset.exerciseStage=String(current);content.dataset.exerciseMode=['intro','objective','scenario','assembled','axis','interactive'][current];
    story.hidden=current===0;story.children[0].hidden=current===0||current===2;story.children[1].hidden=current===0||current===1;
    diagramTitle.hidden=current<4;diagram.hidden=current<4;mobileAxis.hidden=current<4;mobile.hidden=current<5;
    content.setAttribute('role',current<5?'button':'group');content.tabIndex=current<5?0:-1;content.setAttribute('aria-disabled',String(!ready));
    entries.forEach(({trigger,causes,button,list,branch},index)=>{const open=current>=5&&Boolean(state.exerciseGroups?.[index]);trigger.toggleAttribute('hidden',current<5);trigger.tabIndex=current>=5?0:-1;branch.toggleAttribute('hidden',current<5);trigger.setAttribute('aria-expanded',String(open));trigger.setAttribute('aria-disabled',String(!ready||open));causes.toggleAttribute('hidden',!open);causes.setAttribute('aria-hidden',String(!open));button.disabled=!ready||current<5;button.setAttribute('aria-expanded',String(open));list.hidden=!open;});
  }
  function advance(){
    if(!ready||stage()>=5||host?.eiuSectionEnded)return;
    if(stage()>0)[title,...story.children].forEach(node=>node.getAnimations?.().forEach(animation=>animation.cancel()));
    const current=stage(),anchor=current===0?title:current===2?story.children[1]:null;
    const animate=anchor&&typeof anchor.animate==='function'&&!host?.matchMedia('(prefers-reduced-motion:reduce)').matches;
    const before=animate?anchor.getBoundingClientRect().top:0;
    state.exerciseStage=current+1;update();content.scrollTop=0;
    if(animate){const distance=before-anchor.getBoundingClientRect().top;if(distance>0)anchor.animate([{transform:`translateY(${distance}px)`},{transform:'translateY(0)'}],{duration:450,easing:'cubic-bezier(.22,1,.36,1)'});}
    save();
  }
  content.addEventListener('click',advance);
  content.addEventListener('keydown',event=>{if(event.target===content&&(event.key==='Enter'||event.key===' ')){event.preventDefault();advance();}});
  function save(){if(requestId&&epoch)host.parent.postMessage({type:'eiu-reveal-save',requestId,epoch,state},'*');}
  if(requestId){const receive=event=>{const value=event.data;if(event.source!==host.parent||value?.type!=='eiu-reveal-state'||value.requestId!==requestId||host.eiuSectionEnded)return;const groups=value.state?.exerciseGroups;if(typeof value.epoch!=='string'||!value.state||(value.state.exerciseStage!==undefined&&(!Number.isInteger(value.state.exerciseStage)||value.state.exerciseStage<0||value.state.exerciseStage>5))||(groups!==undefined&&(!Array.isArray(groups)||groups.length!==6||!groups.every(item=>typeof item==='boolean'))))return;state=value.state;epoch=value.epoch;ready=true;update();};const refresh=()=>{ready=false;update();};host.addEventListener('eiu-reveal-refresh',refresh);host.addEventListener('message',receive);host.addEventListener('pagehide',()=>{host.removeEventListener('message',receive);host.removeEventListener('eiu-reveal-refresh',refresh);},{once:true});host.parent.postMessage({type:'eiu-reveal-load',requestId},'*');}
  content.replaceChildren(style,title,story,diagramTitle,diagram,mobileAxis,mobile);update();content.parentElement.setAttribute('role','group');
}
