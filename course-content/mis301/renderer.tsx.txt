function uuid() {
  const bytes = crypto.getRandomValues(new Uint8Array(16)); bytes[6] = (bytes[6] & 15) | 64; bytes[8] = (bytes[8] & 63) | 128;
  const value = Array.from(bytes, byte => byte.toString(16).padStart(2, '0')).join('');
  return `${value.slice(0,8)}-${value.slice(8,12)}-${value.slice(12,16)}-${value.slice(16,20)}-${value.slice(20)}`;
}
function practiceRequest(operation, body = {}) {
  return new Promise((resolve, reject) => {
    const token = uuid(), requestId = window.eiuLessonRequestId;
    const cleanup = () => { clearTimeout(timer); window.removeEventListener('message', receive); };
    const receive = event => {
      if (event.source !== parent || event.data?.type !== 'eiu-practice-response' || event.data.requestId !== requestId || event.data.token !== token) return;
      cleanup(); const response = event.data.response;
      if (response?.success) resolve(response); else reject(Error(response?.message || 'Unable to complete the request.'));
    };
    const timer = setTimeout(() => { cleanup(); reject(Error('Connection timed out. Your answer is kept; please retry.')); }, 18000);
    window.addEventListener('message', receive);
    parent.postMessage({ type: 'eiu-practice-request', requestId, token, operation, ...body }, '*');
  });
}

    } catch (error) { status.dataset.error = 'true'; status.textContent = error.message; }
    finally { saving = false; valid(); }
  } });
  function valid() { send.disabled = saving || !context || context.teaching || !dirty || !Object.values(inputs).every(input => input.value.trim()); }
  definition.fields.forEach(field => {
    const multiline = /code|explanation/i.test(field.id);
    const input = h(multiline ? 'textarea' : 'input', { type: multiline ? undefined : 'text', required: true, maxLength: 2000, spellcheck: false, 'aria-label': field.label,
      onFocus: () => { if (/code/i.test(field.id)) activeCode = field.id; },
      placeholder: 'Your answer', onInput: () => { dirty = true; progress('draft'); status.textContent = 'Unsaved changes'; status.dataset.error = 'false'; valid(); } });
    inputs[field.id] = input; form.appendChild(h('label', null, field.label, input));
  });
  form.appendChild(send);
  async function load(restore) {
    if (loading || saving) return; loading = true;
    try {
      const next = await practiceRequest('load');
      if (context && context.practiceSession !== next.practiceSession) { dirty = true; status.textContent = 'A new practice session has started. Review your answers before sending.'; }
      context = next;
      window.dispatchEvent(new CustomEvent('eiu-practice-context', { detail: next }));
      if (next.teaching) {
        heading.textContent = 'Practice responses';
        identity.textContent = 'Teaching view · latest 200 submissions'; form.hidden = true; list.hidden = false;
        const items = next.items.filter(item => item.partId === slide.practicePart);
        list.replaceChildren(...(items.length ? items.map(item => h('div', { className: 'submission' }, h('strong', null, item.name), h('small', null, new Date(item.submittedAt).toLocaleString()),
          ...definition.fields.map(field => h('pre', null, field.label + ': ' + (item.values?.[field.id] || ''))))) : [h('p', { className: 'support' }, 'No answers sent for this part yet.')]));
      } else {
        identity.textContent = `${next.learner.name} · ${next.learner.email}`; list.hidden = true; form.hidden = false;
        const previous = next.items.find(item => item.partId === slide.practicePart);
        if (restore && !dirty && previous) { Object.entries(inputs).forEach(([key, input]) => { input.value = previous.values?.[key] || ''; }); status.textContent = 'Your last submitted answer is restored.'; }
      }
      valid();
    } catch (error) { identity.textContent = 'Learning Access required'; status.dataset.error = 'true'; status.textContent = error.message; }
    finally { loading = false; }
  }
  const refresh = h('button', { type: 'button', onClick: () => load(true) }, 'Refresh / reconnect');
  window.addEventListener('message', event => { if (visible() && event.source === parent && event.data?.type === 'eiu-practice-changed' && event.data.requestId === window.eiuLessonRequestId && context?.teaching) void load(false); });
  window.addEventListener('message', event => { if (visible() && event.source === parent && event.data?.type === 'eiu-classroom-state' && event.data.requestId === window.eiuLessonRequestId && event.data.snapshot?.ready && context) void load(false); });
  window.addEventListener('eiu-practice-activate', event => { if (event.detail.partId === slide.practicePart && visible()) void load(true); });
  window.addEventListener('eiu-use-cell', event => {
    if (event.detail.partId !== slide.practicePart || !visible()) return;
    const field = activeCode || Object.keys(inputs).find(key => /code/i.test(key) && !inputs[key].value) || Object.keys(inputs).find(key => /code/i.test(key));
    if (field) { inputs[field].value = event.detail.code; dirty = true; progress('draft'); status.textContent = 'Code copied · complete the remaining answers'; valid(); }
  });
  void load(true);
  panel.append(heading, identity, form, list, status, refresh);
  return panel;
}

function download(name) {
  const bytes = Uint8Array.from(atob(assets[name]), char => char.charCodeAt(0));
  const type = name.endsWith('.csv') ? 'text/csv;charset=utf-8' : name.endsWith('.ipynb') ? 'application/x-ipynb+json' : 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet';
  const url = URL.createObjectURL(new Blob([bytes], { type }));
  const a = document.createElement('a'); a.href = url; a.download = name; a.click();
  setTimeout(() => URL.revokeObjectURL(url), 1000);
}

const numeric = value => typeof value === 'number' && Number.isFinite(value);
const format = value => value == null ? 'NaN' : typeof value === 'number' && !Number.isInteger(value) ? value.toFixed(6) : String(value);
function table(columns, rows, indices) {
  return h('table', null, h('thead', null, h('tr', null, h('th', null, ''), columns.map(column => h('th', null, column)))),
    h('tbody', null, rows.map((row, i) => h('tr', null, h('td', null, indices?.[i] ?? i), row.map(value => h('td', null, format(value)))))));
}
function preview(data, start, count, full = false) {
  const rows = data.rows.slice(start, start + count), indices = rows.map((_, i) => start + i);
  if (full && data.rows.length > 10) { rows.splice(5, rows.length - 5, ...data.rows.slice(-5)); indices.splice(5, indices.length - 5, ...data.rows.slice(-5).map((_, i) => data.rows.length - 5 + i)); }
  return h('div', null, table(data.columns, rows, indices), h('p', { className: 'dfnote' }, full ? `${data.rows.length} rows × ${data.columns.length} columns · first/last 5 shown` : `${rows.length} rows × ${data.columns.length} columns`));
}
function describe(data) {
  const columns = data.columns.map((name, i) => ({ name, values: data.rows.map(row => row[i]).filter(numeric).sort((a, b) => a - b) })).filter(column => column.values.length);
  const labels = ['count', 'mean', 'std', 'min', '25%', '50%', '75%', 'max'];
  const results = columns.map(({ values: v }) => {
    const mean = v.reduce((a, b) => a + b, 0) / v.length;
    const q = p => { const index = (v.length - 1) * p, low = Math.floor(index); return v[low] + (v[Math.ceil(index)] - v[low]) * (index - low); };
    return [v.length, mean, v.length > 1 ? Math.sqrt(v.reduce((sum, x) => sum + (x - mean) ** 2, 0) / (v.length - 1)) : null, v[0], q(.25), q(.5), q(.75), v[v.length - 1]];
  });
  return table(columns.map(column => column.name), labels.map((_, i) => results.map(result => result[i])), labels);
}
function info(data) {
  const lines = ["<class 'pandas.core.frame.DataFrame'>", `RangeIndex: ${data.rows.length} entries, 0 to ${data.rows.length - 1}`, `Data columns (total ${data.columns.length} columns):`, ' #  Column                          Non-Null Count  Dtype'];
  data.columns.forEach((column, i) => {
    const v = data.rows.map(row => row[i]).filter(value => value !== null);
    const type = v.every(numeric) ? v.length === data.rows.length && v.every(Number.isInteger) ? 'int64' : 'float64' : 'object';
    lines.push(` ${i}  ${column.padEnd(32)} ${String(v.length).padEnd(5)} non-null  ${type}`);
  });
  return h('pre', null, lines.join('\n'));
}
function Notebook(slide) {
  // This is a finite command interpreter over bundled data, never eval/Python.
  let uploaded = Boolean(slide.bootstrap), imported = Boolean(slide.bootstrap), execution = 0, busy = false;
  let selected = 0, frames = slide.bootstrap ? { df: datasets[slide.dataset][0] } : {};
  const output = h('div', { className: 'output', role: 'status', 'aria-live': 'polite' }, h('p', { className: 'support' }, slide.bootstrap ? 'The file, Pandas and df are ready for this inspection exercise.' : '1. Upload the lesson file. 2. Run the cells in order.'));
  const file = h('div', { className: 'fileitem' }, uploaded ? '✓ ' + slide.dataset : 'No file uploaded');
  const status = h('div', { className: 'execution' }, 'Not executed');
  const editor = h('textarea', { value: slide.commands[0], spellcheck: false, 'aria-label': 'Python demo cell' });
  const replace = content => output.replaceChildren(h('div', { className: 'cell-result' }, content));
  function evaluate(source) {
    const lines = source.split('\n').map(line => line.trim()).filter(line => line && !line.startsWith('#'));
    if (!lines.length) return h('pre', null, '');
    // A single cell is atomic only in presentation; prior successful statements remain in runtime state.
    let result = h('pre', null, '');
    for (const line of lines) {
      if (line === 'import pandas as pd') { imported = true; result = h('p', { className: 'status' }, 'Pandas imported. This cell has no display output.'); continue; }
      const read = line.match(/^(?:(df2?)\s*=\s*)?pd\.read_(csv|excel)\(\s*['"]([^'"]+)['"]\s*(?:,\s*sheet_name\s*=\s*(\d+|['"][^'"]+['"]))?\s*\)$/);
      if (read) {
        if (!imported) throw Error("NameError: name 'pd' is not defined. Run import pandas as pd first.");
        const name = read[3].replace(/^\/content\//, '');
        if (!uploaded || name !== slide.dataset) throw Error(`FileNotFoundError: ${name}. Upload the file for this part first.`);
        if ((read[2] === 'csv') !== name.endsWith('.csv')) throw Error('Use read_csv for CSV and read_excel for the Excel workbook.');
        const worksheets = datasets[name], target = read[4];
        const data = target && /['"]/.test(target) ? worksheets.find(sheet => sheet.name === target.slice(1, -1)) : worksheets[Number(target || 0)];
        if (!data) throw Error('ValueError: worksheet not found. Use sheet_name=0 or sheet_name=1.');
        if (read[1]) { frames[read[1]] = data; result = h('p', { className: 'status' }, `${read[1]} created. Assignment has no display output; run ${read[1]}.head() to inspect it.`); }
        else result = preview(data, 0, data.rows.length, true);
        continue;
      }
      const query = line.match(/^(df2?)(?:\.(shape|head|tail|count|describe|info)(?:\((\d*)\))?(?:\[(0|1)\])?|\[['"]([^'"]+)['"]\]\.(mean|min|count)\(\))?$/);
      if (!query) throw Error('Demo supports the command chips, df / df2, head(n), tail(n), shape, count(), describe(), info(), and column mean()/min()/count(). Run other Python in the downloaded notebook.');
      const data = frames[query[1]]; if (!data) throw Error(`NameError: name '${query[1]}' is not defined. Run the import and read cell first.`);
      const method = query[2];
      if (query[5]) {
        const index = data.columns.indexOf(query[5]); if (index < 0) throw Error(`KeyError: '${query[5]}'`);
        const values = data.rows.map(row => row[index]).filter(value => value !== null);
        if (query[6] !== 'count' && !values.every(numeric)) throw Error('TypeError: this column is not numeric.');
        result = h('pre', null, query[6] === 'count' ? String(values.length) : query[6] === 'min' ? format(values.reduce((a, b) => Math.min(a, b), Infinity)) : format(values.reduce((a, b) => a + b, 0) / values.length));
      } else if (!method) result = preview(data, 0, data.rows.length, true);
      else if (method === 'shape') result = h('pre', null, query[4] ? String(query[4] === '0' ? data.rows.length : data.columns.length) : `(${data.rows.length}, ${data.columns.length})`);
      else if (method === 'head' || method === 'tail') { const n = Math.min(data.rows.length, Number(query[3] || 5)); result = preview(data, method === 'tail' ? data.rows.length - n : 0, n); }
      else if (method === 'count') result = h('pre', null, data.columns.map((column, i) => `${column.padEnd(38)} ${data.rows.filter(row => row[i] !== null).length}`).join('\n') + '\ndtype: int64');
      else if (method === 'info') result = info(data);
      else result = describe(data);
    }
    return result;
  }
  const run = h('button', { className: 'run', type: 'button', 'aria-label': 'Run cell', onClick: () => {
    if (busy) return; busy = true; run.disabled = true; editor.disabled = true;
    run.classList.add('busy'); status.textContent = 'Running…';
    setTimeout(() => {
      try { replace(evaluate(editor.value)); status.textContent = `[${++execution}] Executed · offline demo`; if (editor.value.trim() === slide.commands[selected].trim()) buttons[selected].classList.add('executed'); }
      catch (error) { replace(h('pre', { className: 'error' }, error.message)); status.textContent = `[${++execution}] Check the cell`; }
      busy = false; run.disabled = false; editor.disabled = false; run.classList.remove('busy');
    }, 240);
  } }, '▶');
  editor.addEventListener('keydown', event => { if (event.shiftKey && event.key === 'Enter') { event.preventDefault(); run.click(); } });
  function selectCell(index) { if (busy) return; selected = index; editor.value = slide.commands[index]; buttons.forEach((button, i) => button.setAttribute('aria-pressed', String(index === i))); editor.focus(); }
  const buttons = slide.commands.map((command, i) => h('button', { type: 'button', 'aria-pressed': i === selected, onClick: () => selectCell(i) }, command));
  const toolbar = h('div', { className: 'cell-tools' },
    h('button', { type: 'button', onClick: () => selectCell((selected + 1) % slide.commands.length) }, 'Next cell →'),
    slide.practicePart && h('button', { type: 'button', onClick: () => window.dispatchEvent(new CustomEvent('eiu-use-cell', { detail: { partId: slide.practicePart, code: editor.value } })) }, 'Use code in answer'),
    h('button', { type: 'button', onClick: () => { if (busy) return; uploaded = imported = Boolean(slide.bootstrap); execution = 0; frames = slide.bootstrap ? { df: datasets[slide.dataset][0] } : {}; buttons.forEach(button => button.classList.remove('executed')); selectCell(0); file.textContent = uploaded ? '✓ ' + slide.dataset : 'No file uploaded'; status.textContent = 'Runtime reset'; replace(h('p', { className: 'support' }, 'Ready for a fresh run.')); } }, 'Reset runtime'));
  return h('div', { className: slide.practicePart ? 'lab practice-lab' : 'lab' }, h('div', { className: 'notebook' },
    h('div', { className: 'nbbar' }, h('span', { className: 'colabmark' }, 'co'), h('strong', null, slide.phase.startsWith('DEMO') ? 'Demo_Pandas.ipynb' : 'Practice_Pandas.ipynb'), h('small', null, 'COLAB-STYLE · OFFLINE SIMULATION')),
    h('div', { className: 'nbcontent' }, h('aside', { className: 'files' }, h('strong', null, 'Files / content'),
      h('button', { type: 'button', onClick: () => { uploaded = true; file.textContent = '✓ ' + slide.dataset; } }, '↑ Upload lesson file'), file,
      h('p', null, 'Upload loads the bundled file for this part. State resets when leaving the slide.')),
      h('div', { className: 'workspace' }, slide.practicePart && h('div', { className: 'practice-commands' }, buttons), h('div', { className: 'cell' }, run, editor), toolbar, status, output))),
    h('aside', { className: slide.practicePart ? 'practice-side' : 'side' }, !slide.practicePart && h('div', { className: 'tip' }, h('h3', null, 'Choose a cell → Run'), h('div', { className: 'commands' }, buttons)),
      ...(slide.practicePart ? [PracticePanel(slide)] : slide.notes.map((note, i) => h('div', { className: 'tip' }, h('h3', null, String(i + 1).padStart(2, '0')), h('p', null, note)))),
      !slide.practicePart && h('p', { className: 'support' }, 'Shift + Enter runs the cell. Supported commands use the supplied data; this is not a live Python runtime.')));
}
