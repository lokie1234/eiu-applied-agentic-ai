function uuid() {
  const bytes = crypto.getRandomValues(new Uint8Array(16)); bytes[6] = (bytes[6] & 15) | 64; bytes[8] = (bytes[8] & 63) | 128;
  const value = Array.from(bytes, byte => byte.toString(16).padStart(2, '0')).join('');
  return `${value.slice(0,8)}-${value.slice(8,12)}-${value.slice(12,16)}-${value.slice(16,20)}-${value.slice(20)}`;
}

  });
}
async function download(name) {
  const rawCsv = datasets[name]?.[0]?.csvSource;
  let bytes;
  if (rawCsv !== undefined) bytes = new TextEncoder().encode(rawCsv);
  else {
    const asset = assets[name];
    const encoded = typeof asset === 'string' ? asset : asset.gzip;
    bytes = Uint8Array.from(atob(encoded), char => char.charCodeAt(0));
    if (typeof asset !== 'string') bytes = new Uint8Array(await new Response(new Blob([bytes]).stream().pipeThrough(new DecompressionStream('gzip'))).arrayBuffer());
  }
  const type = name.endsWith('.csv') ? 'text/csv;charset=utf-8' : name.endsWith('.ipynb') ? 'application/x-ipynb+json' : 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet';
  const url = URL.createObjectURL(new Blob([bytes], { type }));
  const anchor = document.createElement('a'); anchor.href = url; anchor.download = name; anchor.click();
  setTimeout(() => URL.revokeObjectURL(url), 1000);
}

function describe(data) {
  const columns = data.columns.map((name, i) => ({ name, values: data.rows.map(row => row[i]).filter(numeric).sort((a, b) => a - b) })).filter(column => column.values.length);
  const labels = ['count', 'mean', 'std', 'min', '25%', '50%', '75%', 'max'];
  const results = columns.map(({ values: v }) => {
    const mean = v.reduce((a, b) => a + b, 0) / v.length;
    const q = p => { const index = (v.length - 1) * p, low = Math.floor(index); return v[low] + (v[Math.ceil(index)] - v[low]) * (index - low); };
    return [v.length, mean, v.length > 1 ? Math.sqrt(v.reduce((sum, x) => sum + (x - mean) ** 2, 0) / (v.length - 1)) : null, v[0], q(.25), q(.5), q(.75), v[v.length - 1]];
  });
  return table(columns.map(column => column.name), labels.map((_, i) => results.map(result => result[i])), labels);
}

function ReadCsvDemo(source, separator, header) {
  if (separator.length !== 1 || separator === '\n' || separator === '\r') throw Error('Demo supports a single-character sep such as comma or semicolon.');
  const records = []; let row = [], value = '', quoted = false;
  source = source.replace(/^\uFEFF/, '');
  for (let i = 0; i < source.length; i++) {
    const c = source[i];
    if (c === '"' && quoted && source[i + 1] === '"') { value += '"'; i++; }
    else if (c === '"') quoted = !quoted;
    else if (!quoted && (c === separator || c === '\n')) {
      row.push(value.replace(/\r$/, '')); value = '';
      if (c === '\n') { if (row.some(v => v !== '')) records.push(row); row = []; }
    } else value += c;
  }
  if (quoted) throw Error('ParserError: unterminated quoted field.');
  if (value || row.length) { row.push(value.replace(/\r$/, '')); records.push(row); }
  if (!records.length) throw Error('EmptyDataError: no columns to parse.');
  if (header !== null && header >= records.length) throw Error('ValueError: header row is outside the file.');
  const columns = header === null ? records[0].map((_, i) => String(i)) : records[header].map((v, i) => v || `Unnamed: ${i}`);
  const rows = records.slice(header === null ? 0 : header + 1).map(record => {
    if (record.length > columns.length) throw Error('ParserError: row has more fields than the header. Check sep and header.');
    return columns.map((_, i) => record[i] === undefined || record[i] === '' ? null : Number.isFinite(Number(record[i])) ? Number(record[i]) : record[i]);
  });
  return { columns, rows };
}

    cells.push({ editor, restore: code => { generation++; editor.value = code; result.replaceChildren(); status.textContent = 'Saved code'; }, clear: () => { generation++; result.replaceChildren(); status.textContent = 'Not run'; } });

}

  function captureCell() {

  function evaluate(source) {
    const lines = source.replace(/[A-Za-z_]\w*\.read_(?:csv|excel)\([^)]*\)/g, call => call.replace(/\n\s*/g, ' ')).split('\n').map(line => line.trim()).filter(line => line && !line.startsWith('#'));
    if (!lines.length) return h('pre', null, '');
    // A single cell is atomic only in presentation; prior successful statements remain in runtime state.
    let result = h('pre', null, '');
    for (const line of lines) {
      const importMatch = line.match(/^import pandas(?: as ([A-Za-z_]\w*))?$/);
      if (importMatch) {
        const alias = NotebookIdentifier(importMatch[1] || 'pandas');
        if (!aliases.has(alias) && aliases.size >= 32) throw Error('Demo supports up to 32 Pandas aliases.');
        delete frames[alias]; delete definitions[alias]; aliases.add(alias);
        result = h('p', { className: 'status' }, `Pandas imported as ${alias}.`); continue;
      }
      const match = line.match(/^(?:(?<target>[A-Za-z_]\w*)\s*=\s*)?(?<alias>[A-Za-z_]\w*)\.read_(?<format>csv|excel)\(\s*['"](?<file>[^'"]+)['"](?<args>.*)\)$/);
      const read = match && [match[0], match.groups.target, match.groups.format, match.groups.file, match.groups.args];
      if (read) {
        const alias = match.groups.alias;
        if (!aliases.has(alias)) throw Error(alias in frames ? `AttributeError: '${alias}' is a DataFrame, not the Pandas module.` : `NameError: name '${alias}' is not defined. Import pandas as ${alias} first.`);
        const name = read[3].replace(/^\/content\//, '');
        if (!loadedFiles.has(name)) throw Error(`FileNotFoundError: ${name}. Load the lesson file first.`);
        if ((read[2] === 'csv') !== name.endsWith('.csv')) throw Error('Use read_csv for CSV and read_excel for the Excel workbook.');
        const options = {}; const remainder = read[4].replace(/,\s*(sep|header|sheet_name)\s*=\s*("[^"]*"|'[^']*'|None|\d+)/g, (_, key, value) => {
          if (Object.prototype.hasOwnProperty.call(options, key)) throw Error(`SyntaxError: repeated keyword argument ${key}.`);
          options[key] = value; return '';
        });
        if (remainder.trim()) throw Error('Demo supports named sep, header and sheet_name arguments. Use the example cells.');
        let data, definition = { file: name, format: read[2] };
        if (read[2] === 'csv') {
          if ('sheet_name' in options) throw Error('TypeError: read_csv does not accept sheet_name.');
          const sep = options.sep === undefined ? ',' : /^['"]/.test(options.sep) ? options.sep.slice(1, -1) : '';
          const header = options.header === 'None' ? null : options.header === undefined ? 0 : /^\d+$/.test(options.header) ? Number(options.header) : NaN;
          if (Number.isNaN(header)) throw Error('ValueError: header must be a non-negative integer or None.');
          definition = { ...definition, sep, header };
          const raw = (slide.csvFiles || {})[name] ?? datasets[name]?.[0]?.csvSource;
          data = raw === undefined ? datasets[name][0] : ReadCsvDemo(raw, sep, header);
        } else {
          if ('sep' in options || 'header' in options) throw Error('This Excel demo supports sheet_name only.');
          const worksheets = datasets[name], target = options.sheet_name;
          data = target && /['"]/.test(target) ? worksheets.find(sheet => sheet.name === target.slice(1, -1)) : worksheets[Number(target || 0)];
          if (!data) throw Error('ValueError: worksheet not found. Use sheet_name=0 or sheet_name=1.');
          definition.sheet = target && /['"]/.test(target) ? target.slice(1, -1) : Number(target || 0);
        }
        if (read[1]) { bindFrame(read[1], data, definition); result = h('p', { className: 'status' }, `${read[1]} created. Assignment has no display output; run ${read[1]} to display it.`); }
        else result = preview(data, 0, data.rows.length, true);
        continue;
      }
      const assignment = line.match(/^([A-Za-z_]\w*)\s*=\s*([A-Za-z_]\w*)$/);
      if (assignment) {
        const name = NotebookIdentifier(assignment[1]), source = assignment[2];
        if (source in frames) bindFrame(name, frames[source], definitions[source] || {});
        else if (aliases.has(source)) { if (!aliases.has(name) && aliases.size >= 32) throw Error('Demo supports up to 32 Pandas aliases.'); delete frames[name]; delete definitions[name]; aliases.add(name); }
        else throw Error(`NameError: name '${source}' is not defined.`);
        result = h('p', { className: 'status' }, `${name} assigned.`); continue;
      }
      const query = line.match(/^([A-Za-z_]\w*)(?:\.(shape|head|tail|count|describe|info)(?:\((\d*)\))?(?:\[(0|1)\])?|\[['"]([^'"]+)['"]\]\.(mean|min|count|dtype)(?:\(\))?)?$/);
      if (!query) throw Error('Demo supports Pandas import/read, DataFrame variables, head(n), tail(n), shape, count(), describe(), info(), and column mean()/min()/count(). Run other Python in the downloaded notebook.');
      if (aliases.has(query[1])) {
        if (query[2] || query[5]) throw Error(`AttributeError: '${query[1]}' is the Pandas module, not a DataFrame.`);
        result = h('pre', null, "<module 'pandas'>"); continue;
      }
      const data = frames[query[1]]; if (!data) throw Error(`NameError: name '${query[1]}' is not defined. Run the import and read cell first.`);
      const method = query[2];
      if (method === 'shape' && /\.shape\(/.test(line)) throw Error("TypeError: 'tuple' object is not callable. Use .shape without parentheses.");
      if (method && method !== 'shape' && !line.includes('(')) throw Error(`Use .${method}() to call this method.`);
      if (query[5]) {
        const index = data.columns.indexOf(query[5]); if (index < 0) throw Error(`KeyError: '${query[5]}'`);
        const values = data.rows.map(row => row[index]).filter(value => value !== null);
        if (query[6] === 'dtype') {
          if (/\.dtype\(/.test(line)) throw Error('TypeError: dtype is a property. Use .dtype without parentheses.');
          result = h('pre', null, values.every(numeric) ? values.length === data.rows.length && values.every(Number.isInteger) ? 'int64' : 'float64' : 'object'); continue;
        }
        if (!line.endsWith('()')) throw Error(`Use .${query[6]}() to call this method.`);
        if (query[6] !== 'count' && !values.every(numeric)) throw Error('TypeError: this column is not numeric.');
        result = h('pre', null, query[6] === 'count' ? String(values.length) : query[6] === 'min' ? format(values.reduce((a, b) => Math.min(a, b), Infinity)) : format(values.reduce((a, b) => a + b, 0) / values.length));
      } else if (!method) result = preview(data, 0, data.rows.length, true);
      else if (method === 'shape') result = h('pre', null, query[4] ? String(query[4] === '0' ? data.rows.length : data.columns.length) : `(${data.rows.length}, ${data.columns.length})`);
      else if (method === 'head' || method === 'tail') { const n = Math.min(data.rows.length, Number(query[3] || 5)); result = preview(data, method === 'tail' ? data.rows.length - n : 0, n); }
      else if (method === 'count') result = h('pre', null, data.columns.map((column, i) => `${column.padEnd(38)} ${data.rows.filter(row => row[i] !== null).length}`).join('\n') + '\ndtype: int64');
      else if (method === 'info') result = info(data);
      else result = describe(data);
    }
    return result;
  }
  if (slide.stackedCells) return StackedPracticeNotebook(slide, evaluate, () => {
    aliases = new Set(slide.bootstrap ? ['pd'] : []);

      checkpoint(); refreshBindings();
      busy = false; run.disabled = false; editor.disabled = false; run.classList.remove('busy'); updateGroupUI();
    }, 240);

  editor.addEventListener('keydown', event => { if (event.shiftKey && event.key === 'Enter') { event.preventDefault(); run.click(); } });

    if (busy || (slide.sharedRuntime && !runtimeReady)) return;

    if (previousGroup) groupSelections[previousGroup.start] = selected;

    updateGroupUI(); checkpoint(); slide.onCellSelected?.(index); editor.focus();
  }
  editor.addEventListener('input', () => checkpoint());
  const buttons = slide.commands.map((command, i) => h('button', { type: 'button', 'aria-pressed': i === selected, onClick: () => selectCell(i), title: command }, slide.commandLabels?.[i] || command));

  const groupButtons = (slide.commandGroups || []).map((group, index) => h('button', {
    type: 'button', role: 'tab', id: `notebook-tab-${index}`, 'aria-controls': 'notebook-command-panel',
    onClick: () => selectCell(groupSelections[group.start] ?? group.start)
  }, group.label));
  const targetSelect = h('select', { 'aria-label': 'DataFrame to explore', onChange: () => {
    if (busy || !runtimeReady) { updateGroupUI(); return; }
    const group = groupFor(selected), target = group?.targets?.find(target => target.value === targetSelect.value);
    const current = group?.targets?.find(target => selected >= target.start && selected < target.start + target.count);
    if (target && current) selectCell(target.start + selected - current.start);
  } }, ['df', 'df2'].map(name => h('option', { value: name }, name === 'df' ? 'df - CSV' : 'df2 - Excel')));
  const targetControl = h('label', { className: 'workflow-target' }, h('span', null, 'DataFrame'), targetSelect);
  const groupNav = h('div', { className: 'workflow-nav' }, h('div', { className: 'workflow-tabs', role: 'tablist', 'aria-label': 'Pandas workflow', onKeyDown: event => {
    const current = groupButtons.indexOf(document.activeElement);
    if (current < 0 || !['ArrowLeft', 'ArrowRight', 'Home', 'End'].includes(event.key)) return;
    event.preventDefault();
    const next = event.key === 'Home' ? 0 : event.key === 'End' ? groupButtons.length - 1 : (current + (event.key === 'ArrowRight' ? 1 : -1) + groupButtons.length) % groupButtons.length;
    groupButtons[next].click(); groupButtons[next].focus();
  } }, groupButtons), targetControl);
  function updateGroupUI() {
    if (!slide.commandGroups) return;
    const group = groupFor(selected) || slide.commandGroups[0];
    const target = group.targets?.find(target => selected >= target.start && selected < target.start + target.count);

    groupButtons.forEach((button, index) => {

    });

    buttons.forEach((button, index) => { button.hidden = index < visible.start || index >= visible.start + visible.count; });
    presets.setAttribute('data-count', String(visible.count));
    presets.setAttribute('role', 'tabpanel'); presets.id = 'notebook-command-panel';
    presets.setAttribute('aria-labelledby', `notebook-tab-${slide.commandGroups.indexOf(group)}`);
  }
  const toolbar = h('div', { className: 'cell-tools' },
    h('button', { type: 'button', onClick: () => selectCell((selected + 1) % slide.commands.length) }, 'Next cell →'),
);


  const notebook = h('div', { className: 'lab' }, h('div', { className: 'notebook' },
    h('div', { className: 'nbbar' }, h('span', { className: 'colabmark' }, 'co'), h('strong', null, slide.phase.startsWith('DEMO') ? 'Demo_Pandas.ipynb' : 'Practice_Pandas.ipynb')),

      !slide.autoUpload && h('button', { type: 'button', onClick: () => { uploaded = true; loadedFiles.add(slide.dataset); Object.keys(slide.csvFiles || {}).forEach(name => loadedFiles.add(name)); refreshBindings(); checkpoint(); } }, 'Upload lesson file'), file,

);

  refreshBindings(); updateGroupUI();
  if (slide.sharedRuntime) {
    window.addEventListener('message', event => {
      if (event.source !== parent || event.data?.type !== 'eiu-notebook-terminated' || event.data.requestId !== window.eiuLessonRequestId) return;
      notebookTerminated = true; runtimeReady = false;
      notebook.querySelectorAll('button,textarea,select').forEach(control => { control.disabled = true; });
      status.textContent = 'Section terminated';
      replace(h('p', { className: 'support' }, 'This notebook session has ended. Reopen the section to start a new session.'));
    });
    status.textContent = 'Restoring notebook session...';
    LoadNotebookRuntime().then(state => {
      if (notebookTerminated) return;
      runtimeSessionKey = state.sessionKey;
      aliases = new Set(state.aliases); loadedFiles = new Set(state.files);
      if (slide.autoUpload) { loadedFiles.add(slide.dataset); (slide.autoUploadFiles || []).forEach(name => loadedFiles.add(name)); }
      execution = state.execution; frames = Object.create(null); definitions = Object.create(null);
      state.frames.forEach(definition => {
        const data = definition.format === 'csv'
          ? ReadCsvDemo(datasets[definition.file][0].csvSource, definition.sep, definition.header)
          : typeof definition.sheet === 'string' ? datasets[definition.file].find(sheet => sheet.name === definition.sheet) : datasets[definition.file][definition.sheet];
        if (!data) throw Error('Unable to restore the saved worksheet. Reopen this section to retry.');
        bindFrame(definition.name, data, definition);
      });
      pageStates = Object.create(null);
      (state.pages || []).forEach(page => { pageStates[page.page] = page; });
      const savedPage = pageStates[runtimePage];
      cellStates = Object.create(null);
      savedPage?.cells.forEach(cell => { if (cell.index < slide.commands.length) cellStates[cell.index] = cell; });
      selected = savedPage && savedPage.selected < slide.commands.length ? savedPage.selected : 0;
      buttons.forEach((button, index) => {
        button.setAttribute('aria-pressed', String(index === selected));
        button.classList.toggle('executed', Boolean(cellStates[index]?.executed));
      });
      if (savedPage) showCell(selected);
      else { status.textContent = execution ? `[${execution}] Notebook session restored` : 'Ready'; output.replaceChildren(); }
      uploaded = loadedFiles.size > 0; runtimeReady = true; run.disabled = false;
      refreshBindings(); updateGroupUI(); checkpoint();
    }).catch(error => { replace(h('pre', { className: 'error' }, error.message)); status.textContent = 'Session unavailable'; });
  }
  return notebook;
}

function render(slide) {
  const visible = course.slides.filter(item => (window.eiuLessonTeaching || !item.adminOnly) && (window.eiuLessonTeaching || window.eiuPracticeAllowed !== false || item.kind !== 'picker'));
  return Slide(slide, visible.indexOf(slide), visible.length);
}
exports.default = course.slides.map(s => Object.assign(() => render(s), { adminOnly: s.adminOnly === true, drawRestricted: s.kind === 'picker' }));
