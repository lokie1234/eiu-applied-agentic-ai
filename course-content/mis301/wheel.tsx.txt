// Native SVG only. The server chooses the full group once; this view animates it.
function wheelSvg(tag, props = {}, ...children) {
  const node = document.createElementNS('http://www.w3.org/2000/svg', tag);
  for (const [key, value] of Object.entries(props)) if (value != null) node.setAttribute(key === 'className' ? 'class' : key, String(value));
  for (const child of children.flat(Infinity)) if (child != null && child !== false) node.appendChild(child instanceof Node ? child : document.createTextNode(String(child)));
  return node;
}
function WheelGame() {
  const questionSetup = QuestionSetup();
  const s = wheelSvg, reduced = matchMedia('(prefers-reduced-motion: reduce)').matches;
  let demoMode = true, connected = false, members = [...demoLearners], frozen = [], segments = [], rotation = 0, running = false, pending = false, resetting = false, locked = false;
  let requestId = uuid(), drawState = null, latest = null, epoch = null, generation = 0, raf = 0, lit = -1, finishMotion = null, lastTick = 0, tickAnimation = null, previousFrame = 0;
  const disc = h('div', { className: 'wheel-disc' });
  const machine = h('div', { className: 'wheel-machine' });
  const pointer = s('svg', { viewBox: '0 0 30 40', className: 'wheel-pointer', 'aria-hidden': 'true' }, s('path', { d: 'M2 3 Q15 -2 28 3 L15 38 Z', fill: '#F4DFB0', stroke: '#9D7944', 'stroke-width': 2 }));
  const stage = h('div', { className: 'question-stage', role: 'list', tabIndex: 0, 'aria-label': 'Part assignments', 'aria-live': 'polite' });
  let questionSlots = [], numberingVersion = 2;
  const status = h('p', { className: 'wheel-status', role: 'status' }, 'Connecting to the live class…');
  const rosterCount = h('small', null, 'DEMO · 40 learners');
  const modeToggle = h('button', { type: 'button', disabled: true, onClick: async () => {
    if (running || pending || resetting) return;
    try { const response = await practiceRequest('demoMode', { enabled: !demoMode }); update(response); }
    catch (error) { status.dataset.error = 'true'; status.textContent = error.message; }
  } }, 'Exit demo');
  const count = h('input', { type: 'number', value: 1, min: 1, max: 50, 'aria-label': 'Number of learners to draw', onInput: () => { settings.textContent = `Group · ${count.value || '1'} ▾`; } });
  const settingPanel = h('div', { className: 'wheel-setting', hidden: true }, h('label', null, 'People to draw', count), h('button', { type: 'button', onClick: () => { settingPanel.hidden = true; settings.setAttribute('aria-expanded', 'false'); } }, 'Done'));
  const settings = h('button', { type: 'button', 'aria-expanded': 'false', onClick: () => { settingPanel.hidden = !settingPanel.hidden; settings.setAttribute('aria-expanded', String(!settingPanel.hidden)); if (!settingPanel.hidden) count.focus(); } }, 'Group · 1 ▾');
  function welcome() {
    questionSlots = [];
    stage.replaceChildren();
  }
  function prepareQuestions(total, version = 2) {
    numberingVersion = version;
    const columns = total <= 6 ? 1 : total <= 16 ? 2 : total <= 30 ? 3 : total <= 40 ? 4 : 5;
    const rows = Math.ceil(total / columns);
    stage.style.setProperty('--columns', columns); stage.style.setProperty('--rows', rows);
    stage.dataset.columns = String(columns);
    stage.style.setProperty('--name-size', `${columns === 1 ? 2.6 : columns === 2 ? 2 : columns === 3 ? 1.6 : 1.3}vw`);
    questionSlots = Array.from({ length: total }, (_, index) => h('div', { className: 'question-slot', role: 'listitem', hidden: true, style: { gridRow: String(index % rows + 1), gridColumn: String(Math.floor(index / rows) + 1) } }, h('span', { className: 'question-number' }, `Part ${numberingVersion === 2 ? index % 7 + 1 : index % 5 + 3}`), h('strong', { className: 'question-name' }, '?')));
    stage.replaceChildren(...questionSlots); stage.scrollTop = 0;
  }
  function fitQuestion(slot) {
    const title = slot.querySelector('.question-number'), name = slot.querySelector('.question-name');
    title.style.fontSize = ''; name.style.fontSize = '';
    const base = parseFloat(getComputedStyle(name).fontSize);
    let size = base;
    // Keep the title larger while fitting the full username inside its card.
    for (let step = 0; step < 24; step++) {
      title.style.fontSize = `${size * 1.2}px`; name.style.fontSize = `${size}px`;
      if (slot.scrollHeight <= slot.clientHeight + 1 && slot.scrollWidth <= slot.clientWidth + 1) break;
      size *= .94;
    }
  }
  const questionResize = new ResizeObserver(() => questionSlots.forEach(slot => { if (!slot.hidden) fitQuestion(slot); }));
  questionResize.observe(stage);
  window.addEventListener('pagehide', () => questionResize.disconnect(), { once: true });
  function showName(name, index, restored = false) {
    const slot = questionSlots[index]; if (!slot || slot.classList.contains('is-revealed')) return;
    slot.querySelector('.question-name').textContent = name;
    slot.setAttribute('aria-label', `Part ${numberingVersion === 2 ? index % 7 + 1 : index % 5 + 3}: ${name}`);
    slot.title = `Part ${numberingVersion === 2 ? index % 7 + 1 : index % 5 + 3}: ${name}`;
    slot.hidden = false;
    fitQuestion(slot);
    slot.classList.add('is-revealed');
    if (restored) slot.classList.add('is-restored');
    stage.querySelector('.is-current')?.classList.remove('is-current');
    slot.classList.add('is-current');
  }
  function controls() { draw.disabled = !connected || running || pending || resetting || locked || members.length === 0; settings.disabled = running || pending || resetting || locked; reset.disabled = running || pending || resetting || !locked; modeToggle.disabled = !connected || running || pending || resetting; count.disabled = running || locked; drawLabel.textContent = running ? 'Spinning' : locked ? 'Done' : 'Draw'; }
  function rotate(value) {
    const now = performance.now(), speed = Math.abs(value-rotation)/Math.max(1, now-previousFrame)*1000; previousFrame = now;
    rotation = value; disc.style.transform = `rotate(${value}deg)`;
    if (!segments.length) return;
    const index = Math.floor(((360 - (value % 360) + 360) % 360) / (360 / segments.length)) % segments.length;
    if (index !== lit) {
      segments[lit]?.classList.remove('is-lit'); segments[index]?.classList.add('is-lit'); lit = index;
      if (running && !reduced && now-lastTick > 40) {
        lastTick = now; tickAnimation?.cancel();
        const kick = Math.max(4, Math.min(24, speed*.028));
        tickAnimation = pointer.animate([{ transform: `rotate(${-kick}deg)` }, { transform: `rotate(${kick*.3}deg)`, offset: .58 }, { transform: 'rotate(0deg)' }], { duration: Math.max(90, Math.min(210, 240-speed*.1)), easing: 'cubic-bezier(.2,.8,.3,1)' });
      }
    }
  }
  function renderWheel() {
    const pool = running ? frozen : members, n = pool.length, center = 200, r = 176;
    const svg = s('svg', { viewBox: '-6 -6 412 412', width: '100%', height: '100%', role: 'img', 'aria-label': `${n} learner names on the wheel` });
    segments = []; lit = -1;
    if (!n) svg.appendChild(s('circle', { cx: center, cy: center, r, fill: '#123F68' }));
    pool.forEach((member, i) => {
      const step = 360 / n, a = (-90 + i * step) * Math.PI / 180, b = (-90 + (i + 1) * step) * Math.PI / 180;
      const colors = ['#123F68','#18547F','#276A91','#0e3556'];
      const sector = n === 1 ? s('circle', { cx: center, cy: center, r, fill: colors[0], className: 'sector' }) : s('path', { d: `M200 200 L${center + r * Math.cos(a)} ${center + r * Math.sin(a)} A${r} ${r} 0 ${step > 180 ? 1 : 0} 1 ${center + r * Math.cos(b)} ${center + r * Math.sin(b)} Z`, fill: colors[i % colors.length], stroke: '#ffffff35', 'stroke-width': .6, className: 'sector' });
      const size = n > 30 ? 6.5 : n > 18 ? 8 : 10;
      const angle = -90 + (i + .5) * step;
      // Tapered plates stay strictly inside their own sector instead of overlapping
      // adjacent rectangles near the hub when 40–50 names are present.
      const inner = 80, outer = 169, half = step * .36 * Math.PI / 180;
      const point = (radius, theta) => `${200 + radius * Math.cos(theta)} ${200 + radius * Math.sin(theta)}`;
      const plate = n === 1 ? s('rect', { x: 276, y: 200 - size, width: 94, height: size * 1.8, rx: 4, className: 'name-plate' }) :
        s('path', { d: `M${point(inner,-half)} L${point(outer,-half)} A${outer} ${outer} 0 0 1 ${point(outer,half)} L${point(inner,half)} A${inner} ${inner} 0 0 0 ${point(inner,-half)} Z`, className: 'name-plate' });
      const maxCharacters = Math.max(12, Math.floor(81 / (size * .58)));
      const display = member.username.length > maxCharacters ? member.username.slice(0, maxCharacters - 1) + '…' : member.username;
      const group = s('g', { className: 'wheel-segment' }, s('title', {}, member.username), sector,
        s('g', { transform: `rotate(${angle} 200 200)` }, plate, s('text', { x: 365, y: 200 + size * .3, 'text-anchor': 'end', 'font-size': size }, display)));
      segments.push(group); svg.appendChild(group);
    });
    disc.replaceChildren(svg); rotate(rotation);
  }
  function memberIndex(pool, state, index) { const idMatch = state.subjects?.[index] ? pool.findIndex(member => member.id === state.subjects[index]) : -1; return idMatch >= 0 ? idMatch : pool.findIndex(member => member.username === state.names[index]); }
  function markPicked(state, index) { const at = memberIndex(running ? frozen : members, state, index); if (at >= 0) segments[at]?.classList.add('is-picked'); return at; }
  function saved(state) {
    drawState = state; locked = true; count.value = state.names.length; settings.textContent = `Group · ${state.names.length} ▾`;
    prepareQuestions(state.names.length, state.assignmentVersion || 1); state.names.forEach((name, index) => { showName(name, index, true); markPicked(state, index); });
    const last = memberIndex(members, state, state.names.length - 1); if (last >= 0) rotate(-(last + .5) * 360 / members.length);
    status.dataset.error = 'false'; status.textContent = `${state.names.length} selected · Reset to start a new draw.`; controls();
  }
  function clear() { generation++; cancelAnimationFrame(raf); tickAnimation?.cancel(); finishMotion?.(); finishMotion = null; running = pending = locked = false; drawState = null; requestId = uuid(); machine.classList.remove('is-spinning'); rotation = 0; renderWheel(); welcome(); status.dataset.error = 'false'; status.textContent = `${members.length} learners ready · names appear after the wheel stops.`; controls(); }
  async function play(state, startTime) {
    const token = ++generation; cancelAnimationFrame(raf); finishMotion?.(); finishMotion = null;
    drawState = state; locked = true; running = true; frozen = [...members];
    renderWheel(); prepareQuestions(state.names.length, state.assignmentVersion || 1); controls();
    const total = state.names.length, targets = [rotation];
    for (let index = 0; index < total; index++) {
      const winner = memberIndex(frozen, state, index);
      const angle = winner < 0 ? 0 : -(winner + .5) * 360 / Math.max(1, frozen.length);
      const previous = targets[index];
      targets.push(previous + 360 + ((angle - previous) % 360 + 360) % 360);
    }
    let revealed = 0;
    await new Promise(resolve => {
      finishMotion = resolve;
      const frame = now => {
        if (generation !== token) { resolve(); return; }
        const elapsed = Math.max(0, now - startTime), cycle = Math.min(total, Math.floor(elapsed/2000)), phase = elapsed%2000;
        const visible = Math.min(total, cycle + (phase >= 1750 ? 1 : 0));
        while (revealed < visible) { markPicked(state, revealed); showName(state.names[revealed], revealed); revealed++; }
        if (cycle === total) { rotate(targets[total]); finishMotion = null; resolve(); return; }
        if (phase < 1750) {
          machine.classList.add('is-spinning');
          const t = phase/1750, acceleration = .16;
          // Integrate a short acceleration and a long friction-driven coast.
          // Angular velocity is continuous and falls to zero before reveal.
          const ramp = 2*acceleration/Math.PI, coast = (1-acceleration)/4;
          const q = Math.max(0, (t-acceleration)/(1-acceleration));
          const eased = (t <= acceleration ? ramp*(1-Math.cos(Math.PI*t/(2*acceleration))) : ramp + coast*(1-Math.pow(1-q,4))) / (ramp+coast);
          if (!reduced) rotate(targets[cycle] + (targets[cycle+1]-targets[cycle])*eased);
        } else {
          rotate(targets[cycle+1]); machine.classList.remove('is-spinning');
        }
        status.textContent = `${revealed}/${total}`;
        raf = requestAnimationFrame(frame);
      };
      raf = requestAnimationFrame(frame);
    });
    if (generation !== token) return;
    running = false; machine.classList.remove('is-spinning'); tickAnimation?.cancel();
    renderWheel(); state.names.forEach((_, index) => markPicked(state, index));
    const last = memberIndex(members, state, total-1); if (last >= 0) rotate(-(last+.5)*360/members.length);
    status.textContent = `${total}/${total}`; controls();
  }
  function update(snapshot) {
    if (!snapshot?.ready || !snapshot.teaching) return;
    connected = true;
    const nextMode = typeof snapshot.demoMode === 'boolean' ? snapshot.demoMode : demoMode;
    const nextMembers = nextMode ? demoLearners : snapshot.members;
    if (nextMode !== demoMode) { demoMode = nextMode; members = nextMembers; clear(); }
    snapshot = { ...snapshot, members: nextMembers, draw: demoMode ? snapshot.demoDraw || null : snapshot.draw };
    latest = snapshot;
    if ((epoch && epoch !== snapshot.practiceSession) || (drawState && !snapshot.draw)) { members = snapshot.members; clear(); }
    epoch = snapshot.practiceSession;
    members = snapshot.members; rosterCount.textContent = demoMode ? 'DEMO · 40 learners · no report records' : `${members.length} signed-in learners`; modeToggle.textContent = demoMode ? 'Exit demo' : 'Demo · 40'; count.max = Math.max(1, Math.min(50, members.length));
    if (!running) renderWheel();
    if (!pending && !running && snapshot.draw && snapshot.draw.requestId !== drawState?.requestId) {
      const age = Math.max(0, Date.now() - snapshot.draw.createdAt);
      if (age < snapshot.draw.names.length * 2000) void play(snapshot.draw, performance.now() - age); else saved(snapshot.draw);
    }
    if (!running && drawState) { drawState.names.forEach((_, index) => markPicked(drawState, index)); const last = memberIndex(members, drawState, drawState.names.length - 1); if (last >= 0) rotate(-(last + .5) * 360 / members.length); }
    if (!snapshot.draw && !running && !pending && !resetting) { status.dataset.error = 'false'; status.textContent = members.length ? `${members.length} learners ready · names appear after the wheel stops.` : 'Waiting for learners to sign in to this section.'; }
    controls();
  }
  const drawLabel = h('span', null, 'Draw');
  const draw = h('button', { type: 'button', className: 'wheel-hub', disabled: true, 'aria-label': 'Draw the selected number of learners', onClick: async () => {
    if (running || locked || pending) return;
    const requested = Number(count.value);
    if (!Number.isInteger(requested) || requested < 1 || requested > Math.min(50, members.length)) { status.dataset.error = 'true'; status.textContent = `Choose a group size from 1 to ${Math.min(50, members.length)}.`; return; }
    settingPanel.hidden = true; settings.setAttribute('aria-expanded', 'false'); status.dataset.error = 'false';
    const start = performance.now(), activeRequest = requestId; pending = running = true; frozen = [...members]; controls(); machine.classList.add('is-spinning'); status.textContent = 'Spinning · 5.0 s';
    const coastStart=rotation;
    const coast = now => { if (!pending) return; if (!reduced) rotate(coastStart + 360*(now-start)/1000); raf = requestAnimationFrame(coast); }; raf = requestAnimationFrame(coast);
    try { const response = await practiceRequest('draw', { count: requested, excluded: [], submissionId: activeRequest, demo: demoMode }); if (requestId !== activeRequest) return; pending = false; cancelAnimationFrame(raf); await play(response, performance.now() - Math.max(0, Date.now() - response.createdAt)); }
    catch (error) { if (requestId !== activeRequest) return; pending = running = false; cancelAnimationFrame(raf); machine.classList.remove('is-spinning'); if (latest?.draw?.requestId === requestId) saved(latest.draw); else { status.dataset.error = 'true'; status.textContent = error.message; } controls(); }
  } }, h('small', null, 'EIU · MIS 301'), drawLabel);
  const reset = h('button', { type: 'button', className: 'wheel-reset', disabled: true, onClick: async () => {
    if (running || pending || resetting) return; resetting = true; controls();
    try { await practiceRequest('resetDraw', { demo: demoMode }); clear(); }
    catch (error) { status.dataset.error = 'true'; status.textContent = error.message; }
    finally { resetting = false; controls(); }
  } }, '↺ Reset');
  const metalId = 'wheel-metal-' + uuid();
  const rim = s('svg', { viewBox: '-6 -6 412 412', className: 'wheel-rim', 'aria-hidden': 'true' }, s('defs', {}, s('linearGradient', { id: metalId, x1: '0%', y1: '0%', x2: '75%', y2: '100%' }, s('stop', { offset: '0%', 'stop-color': '#A7C4D9' }), s('stop', { offset: '22%', 'stop-color': '#123F68' }), s('stop', { offset: '47%', 'stop-color': '#35658A' }), s('stop', { offset: '70%', 'stop-color': '#041C31' }), s('stop', { offset: '100%', 'stop-color': '#18547F' }))), s('circle', { cx: 200, cy: 200, r: 193, fill: 'none', stroke: '#041C31', 'stroke-width': 24 }), s('circle', { cx: 200, cy: 200, r: 190, fill: 'none', stroke: `url(#${metalId})`, 'stroke-width': 20 }), s('circle', { cx: 200, cy: 200, r: 201, fill: 'none', stroke: '#276A91', 'stroke-width': 1 }), s('circle', { cx: 200, cy: 200, r: 179, fill: 'none', stroke: '#123F68', 'stroke-width': 3 }),
    ...Array.from({ length: 40 }, (_, i) => { const angle = i * Math.PI * 2 / 40; return s('circle', { cx: 200 + 190 * Math.cos(angle), cy: 200 + 190 * Math.sin(angle), r: 3.3, className: 'wheel-bulb' }); }));
  machine.append(disc, rim, pointer, draw);
  renderWheel(); welcome();
  window.addEventListener('pagehide', () => { generation++; cancelAnimationFrame(raf); tickAnimation?.cancel(); finishMotion?.(); }, { once: true });
  window.addEventListener('message', event => { if (event.source === parent && event.data?.type === 'eiu-classroom-state' && event.data.requestId === window.eiuLessonRequestId) update(event.data.snapshot); });
  void practiceRequest('load').then(response => { update(response); if (!response.ready) status.textContent = 'The live classroom is reconnecting. Learner names will appear here automatically.'; }).catch(error => { status.dataset.error = 'true'; status.textContent = error.message; });
  return h('div', { className: 'wheel-arena' }, h('div', { className: 'wheel-bay' }, machine), h('div', { className: 'wheel-toolbar' }, h('div', { className: 'wheel-actions' }, settings, reset, modeToggle, questionSetup.button), rosterCount, settingPanel), stage, status, questionSetup.dialog);
}
