function SourceOverview(slide) {
  const root = h('div', { className: 'concept-deck source-overview' });
  const headingId = 'source-example-' + uuid();
  let opener = null, inertNodes = [];
  const title = h('h2', { id: headingId });
  const body = h('div', { className: 'source-example-body' });
  const close = () => {
    modal.hidden = true;
    inertNodes.forEach(node => { node.inert = false; }); inertNodes = [];
    opener?.focus();
  };
  const closeButton = h('button', { type: 'button', className: 'concept-close', 'aria-label': 'Close example', onClick: close }, 'Close', h('span', { 'aria-hidden': 'true' }, ' ×'));
  const modal = h('section', { className: 'concept-modal', role: 'dialog', 'aria-modal': 'true', 'aria-labelledby': headingId, hidden: true },
    h('div', { className: 'concept-dialog' }, h('header', { className: 'concept-dialog-header' }, title, closeButton), body));
  modal.addEventListener('keydown', event => {
    if (event.key === 'Escape') { event.preventDefault(); event.stopPropagation(); close(); }
    if (event.key === 'Tab') { event.preventDefault(); closeButton.focus(); }
  });
  const open = (source, format, button) => {
    opener = button; title.textContent = format.title;
    const preview = format.id === 'txt'
      ? h('div', { className: 'format-text-preview' }, h('p', null, 'An scored ', h('strong', null, '85'), '.'), h('p', null, 'Binh scored ', h('strong', null, '92'), '.'), h('p', null, 'Chi scored ', h('strong', null, '78'), '.'))
      : format.id === 'nosql'
      ? h('div', { className: 'format-documents' }, ['An · 85', 'Binh · 92', 'Chi · 78'].map((record, index) => h('div', null, ConceptIcon('document'), h('span', null, h('small', null, `Document ${index + 1}`), h('strong', null, record)))))
      : format.id === 'xlsx'
      ? h('div', { className: 'format-workbook' }, h('div', { className: 'workbook-letters' }, h('span', null, 'A'), h('span', null, 'B')), ConceptSampleTable(), h('span', { className: 'workbook-tab' }, 'Scores'))
      : h('div', { className: format.id === 'html' ? 'format-browser' : 'format-table' }, format.id === 'html' && h('div', { className: 'format-address' }, 'scores.html'), ConceptSampleTable());
    body.replaceChildren(
      h('div', { className: 'example-source' }, h('span', { className: 'concept-label' }, format.inputLabel), format.id === 'xlsx' ? h('div', { className: 'xlsx-example-icon' }, DownloadIcon('xlsx')) : ConceptCode(format.input),
        h('div', { className: 'format-difference' }, h('strong', null, format.differenceTitle), h('p', null, format.difference))),
      h('div', { className: 'example-result' }, h('h3', null, format.previewTitle), preview));
    const frame = root.closest('.frame');
    inertNodes = [cards, ...Array.from(frame?.children || []).filter(node => node !== root.closest('main'))];
    inertNodes.forEach(node => { node.inert = true; });
    modal.hidden = false; closeButton.focus();
  };
  const cards = h('div', { className: 'concept-three' }, slide.sources.map(source => {
    const card = h('section', { className: 'source-open source-card', 'aria-label': source.title },
      SourceIllustration(source.icon),
      h('h2', { className: 'source-title' }, source.title),
      h('span', { className: 'source-points', role: 'list' }, source.bullets.map(text => h('span', { role: 'listitem' }, text))),
      h('div', { className: 'source-formats' }, source.formats.map(format => {
        let button;
        button = h('button', { type: 'button', className: 'format-button', 'aria-haspopup': 'dialog', 'aria-label': `Open ${format.label} example`, onClick: () => open(source, format, button) }, format.label);
        return button;
      })));
    return card;
  }));
  root.append(cards, modal); return root;
}
function StructureOverview(slide) {
  const examples = {
    table: () => ConceptSampleTable(),
    json: () => ConceptCode('[\n  {"student": "An", "score": 85},\n  {"student": "Binh", "score": 92}\n]'),
    prose: () => h('div', { className: 'concept-prose' }, ConceptIcon('document'), h('p', null, '“An scored 85. Binh scored 92 in the next test…”'))
  };
  return h('div', { className: 'concept-deck structure-overview' },
    h('div', { className: 'concept-takeaway' }, ConceptIcon('table'), h('p', null, 'For analysis, data needs ', h('strong', { className: 'concept-emphasis' }, 'rows and columns'), '.')),
    h('div', { className: 'concept-three' }, slide.types.map((type, index) => h('section', { className: 'structure-card' },
      h('div', { className: 'structure-heading' }, ConceptIcon(type.icon), h('h2', null, type.title)),
      h('span', { className: 'structure-tag', 'data-ready': index === 0 ? 'true' : 'false' }, type.tag),
      ConceptBullets(type.bullets), h('div', { className: 'structure-preview' }, examples[type.example]()),
      h('div', { className: 'source-formats' }, type.formats.map(format => h('span', null, format)))))));
}
